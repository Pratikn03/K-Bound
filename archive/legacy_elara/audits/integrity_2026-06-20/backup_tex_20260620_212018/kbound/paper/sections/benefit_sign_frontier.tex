\section{The exact benefit-sign frontier}\label{sec:bsfrontier}

We sharpen the central question from \emph{``is target accuracy identifiable?''}
(answered negatively in general by \textbf{Ben-David et al.}\ 2010 and \textbf{Garg et al.}\ 2022) to the
strictly \emph{comparative} object that gating actually needs: the \emph{sign of the
adaptation benefit}
\[
  \Delta \;:=\; R_T(f_0)-R_T(f_a),\qquad \text{adapt is correct}\iff \Delta>0 .
\]
No prior estimator targets $\sign\Delta$: ATC, DoC, GDE, COT, AETTA and
Agreement-on-the-Line all estimate an \emph{accuracy level} of a \emph{single fixed}
model (\textbf{Garg et al.}, ATC; \textbf{Jiang et al.}, GDE; \textbf{Lu et al.}, COT;
\textbf{Lee et al.}, AETTA; \textbf{Baek et al.}, Agreement-on-the-Line).
We give an exact characterization of when $\sign\Delta$ is identifiable from label-free
observables, prove the governing obstruction is a single scalar that is \emph{necessarily}
present, and show the field's heuristics are exactly its degenerate face.

\paragraph{Setup.} Binary $Y\in\{0,1\}$, $0/1$ loss; $f_0,f_a:\mathcal X\to\{0,1\}$ fixed.
$D=\{x:f_0(x)\neq f_a(x)\}$ is the \emph{disagreement region} (observable). Write
$\eta_a(x)=P_T(Y=f_a(x)\mid x)$ and $\bar a=\E_{\mu\mid D}[\eta_a]$, where $\mu=P_T^X$ is
the (observable) target marginal. The \emph{observables} are $\mu$, the maps $f_0,f_a$, the
labeled source $P_S$, and hence any source-derived score $s(x)$ for $\eta_a(x)$ (e.g.\ a
source-calibrated confidence of $f_a$). Define the \emph{observable margin} and the
\emph{calibration drift on the decision region}
\[
  \begin{aligned}
  M &\;:=\; \E_{\mu\mid D}[\,s\,]-\tfrac12 \quad(\text{observable}),\\
  \gamma &\;:=\; \E_{\mu\mid D}[\,\eta_a-s\,]\quad(\text{unobservable}).
  \end{aligned}
\]

\begin{lemma}[Reduction]\label{lem:reduction}
$\Delta = 2\,\mu(D)\,(\bar a-\tfrac12)$ and $\bar a-\tfrac12 = M+\gamma$; hence, whenever
$\mu(D)>0$,
\[
  \boxed{\;\sign\Delta=\sign(M+\gamma)\;}
\]
\end{lemma}
\begin{proof}
Off $D$, $f_0=f_a$ so the pointwise excess loss $\delta(x)=\ell(f_0,Y)-\ell(f_a,Y)=0$. On
$D$ (binary, $0/1$ loss) exactly one of $f_0,f_a$ equals $Y$, so
$\delta=\mathbf 1[f_a{=}Y]-\mathbf 1[f_0{=}Y]=2\,\mathbf 1[f_a{=}Y]-1$, giving
$\E[\delta\mid D]=2\eta_a-1$ pointwise and $\E_{\mu\mid D}[\delta]=2\bar a-1$. Thus
$\Delta=\E_T[\delta]=\mu(D)\,\E_{\mu\mid D}[\delta]=2\mu(D)(\bar a-\tfrac12)$. Linearity gives
$\bar a-\tfrac12=\E_{\mu\mid D}[\eta_a]-\tfrac12=(\E_{\mu\mid D}[s]-\tfrac12)+\E_{\mu\mid D}[\eta_a-s]=M+\gamma$.
Since $\mu(D)>0$, $\sign\Delta=\sign(M+\gamma)$.
\end{proof}

\begin{theorem}[Exact benefit-sign frontier: sufficiency and necessity \textnormal{(main result)}]\label{thm:frontier}
Fix the observables $(\mu,P_S,f_0,f_a)$ with $\mu(D)>0$, so $M$ is determined.
\begin{enumerate}\itemsep1pt
\item[\textnormal{(i)}] \textbf{(Scalar sufficiency.)} Any two targets with the same
$(M,\gamma)$ have the same $\sign\Delta$; $\gamma$ is a one-dimensional sufficient statistic
for the sign given the observables.
\item[\textnormal{(ii)}] \textbf{(Exact frontier.)} Over the target class
$\mathcal C_\beta=\{P_T:\,|\gamma|\le\beta\}$ consistent with the observables, $\sign\Delta$
is identifiable \emph{iff} $|M|>\beta$, and then $\sign\Delta=\sign M$.
\item[\textnormal{(iii)}] \textbf{(Necessity / minimality.)} For \emph{any} class
$\mathcal C$ of targets consistent with the observables on which $\sign\Delta$ is constant
(hence identifiable), $\mathcal C$ lies entirely on one side of the hyperplane
$\gamma=-M$. Consequently every identifying restriction is equivalent to a one-sided
constraint on $\gamma$: \emph{no assumption that does not constrain $\gamma$ relative to
$-M$ can identify $\sign\Delta$.}
\end{enumerate}
\end{theorem}
\begin{proof}
(i) Immediate from Lemma~\ref{lem:reduction}: $\sign\Delta$ is a function of $M+\gamma$, and
$M$ is fixed by the observables. (ii) If $|M|>\beta$ then for all $|\gamma|\le\beta$,
$\sign(M+\gamma)=\sign M$ (the perturbation cannot cross $0$), so the sign is constant on
$\mathcal C_\beta$ and equals $\sign M$. Conversely if $|M|\le\beta$, choose
$\gamma=+\beta$ and $\gamma=-\beta$: both lie in $\mathcal C_\beta$, and
$M+\beta\ge0$ (since $M\ge-\beta$) while $M-\beta\le0$ (since
$M\le\beta$), so the two admissible targets have $\Delta\ge0$ and $\Delta\le0$
respectively; the sign is not constant, hence not identifiable.
(iii) $\sign\Delta=\sign(M+\gamma)$ with $M$ fixed; constancy of $\sign(M+\gamma)$ on
$\mathcal C$ is, by definition, the statement that $M+\gamma$ does not change sign on
$\mathcal C$, i.e.\ $\gamma$ stays on one side of $-M$.
\end{proof}

\smallskip\noindent\emph{Significance (the paper's main result).} The knowable/unknowable
boundary is pinned to a single inequality $|M|>\beta$ in \emph{observable} quantities; the sole
obstruction is the one-dimensional drift $\gamma$, proved \emph{necessary} by clause~(iii)---not an
artifact of any particular estimator. Corollary~\ref{cor:irreducible} shows the budget $\beta$ is
irreducible, and Proposition~\ref{prop:beta0} shows every prior label-free accuracy heuristic
(ATC, DoC, GDE, COT, AETTA, Agreement-on-the-Line) is the degenerate $\beta{=}0$ face of this
frontier. To our knowledge this is the first exact, two-sided characterization of label-free
benefit-sign identifiability.

\begin{proposition}[Tight minimax impossibility]\label{thm:minimax}
Let a committal rule be any map $g$ from the observables to $\{\textsc{adapt},\textsc{freeze}\}$,
and let its error be $\sup_{P_T\in\mathcal C_\beta}\Pr[g\ \text{wrong about }\sign\Delta]$.
Then the minimax committal error equals
\[
  \inf_g\ \sup_{P_T\in\mathcal C_\beta}\ \Pr[g\ \text{wrong}]
  \;=\;
  \begin{cases} 0,& |M|>\beta,\\[2pt] \tfrac12,& |M|\le\beta.\end{cases}
\]
Hence the three-way rule
$\textsc{adapt}$ if $M>\beta$, $\textsc{freeze}$ if $M<-\beta$, else $\textsc{abstain}$,
is the \emph{unique maximal sound} certificate: it never commits to the wrong sign, and it
abstains exactly on the information-theoretically unknowable set $\{|M|\le\beta\}$.
\end{proposition}
\begin{proof}
For $|M|>\beta$, Theorem~\ref{thm:frontier}(ii) makes the sign certain, so $g\equiv\sign M$
errs with probability $0$. For $|M|\le\beta$, consider the two targets $P^+$ ($\gamma=+\beta$)
and $P^-$ ($\gamma=-\beta$) of Theorem~\ref{thm:frontier}(ii); they differ only in the
conditional law $\eta_a$ on $D$, which is \emph{not} part of the observables, so their
observable laws are identical (total variation $0$). Any $g$ is a function of the
observables and therefore returns the \emph{same} action on $P^+$ and $P^-$, which have
opposite $\sign\Delta$; thus $g$ is wrong on exactly one of them, giving worst-case error
$\ge\tfrac12$, and $\tfrac12$ is achieved by either constant rule. This is the Le Cam
two-point bound $\tfrac12(1-\mathrm{TV})$ with $\mathrm{TV}=0$.
\end{proof}

\begin{corollary}[Irreducibility of the drift budget]\label{cor:irreducible}
$\gamma$ is a functional of the unobservable conditional $\eta_a$ on $D$ and is not
determined by the observable law (Proposition~\ref{thm:minimax} exhibits observationally
identical targets with $\gamma=\pm\beta$). Combined with the necessity clause
Theorem~\ref{thm:frontier}(iii), \emph{no label-free certificate can identify $\sign\Delta$
without a supplied bound on $\gamma$}. The calibration-drift budget $\beta$ is therefore
irreducible: the assumption-free / unlabeled-checkable benefit-sign certificate sought by
prior heuristics provably does not exist.
\end{corollary}

\begin{proposition}[The heuristics are the $\beta{=}0$ face]\label{prop:beta0}
The benefit-sign certificate induced by any source-calibrated confidence estimator (ATC),
by model--dropout disagreement (AETTA), or by agreement slope (Agreement-on-the-Line) is
the plug-in $\sign M$, i.e.\ the $\beta{=}0$ instance of Proposition~\ref{thm:minimax}. It is
sound \emph{iff} $\gamma=0$ (calibration transfers on $D$), and its error equals
$\Pr\!\big[\,|\gamma|>|M|\ \text{and}\ \sign\gamma\neq\sign M\,\big]$. Thus these methods do
not fail for independent reasons; they share the single failure mode $\gamma\neq0$, the
calibration drift on the decision region.
\end{proposition}
\begin{proof}
Each method thresholds an observable surrogate $s$ for $\eta_a$ and predicts the benefit
sign by whether the average surrogate exceeds chance, i.e.\ by $\sign(\E_{\mu\mid D}[s]-\tfrac12)=\sign M$;
this is Proposition~\ref{thm:minimax} with $\beta=0$. Correctness $\sign M=\sign(M+\gamma)$ fails
exactly when $\gamma$ has the opposite sign to $M$ and $|\gamma|>|M|$ (Lemma~\ref{lem:reduction}).
\end{proof}

\paragraph{Multiclass and regression.} The reduction generalizes: for $K$-class $0/1$ loss
$\gamma=\E_{\mu\mid D}[(p_a-p_0)-(s_a-s_0)]$ and for squared loss $\gamma$ is the
unobservable moment $\E_{\mu\mid D}[(f_0-f_a)(\,\E[Y\mid x]-\hat m(x))]$ with $\hat m$ the
source regressor; Theorem~\ref{thm:frontier} and Proposition~\ref{thm:minimax} and
Corollary~\ref{cor:irreducible} hold verbatim with the corresponding $\gamma$
(Appendix~\ref{app:theory-ext}).

\paragraph{What this is, and is not.} The impossibility \emph{core}
(\,$\sign\Delta$ unidentifiable in general\,) is known
(\textbf{Ben-David et al.}\ 2010; \textbf{Garg et al.}\ 2022), as is the folklore that
calibration is not unlabeled-checkable (\textbf{Rosenfeld \& Garg}\ 2023). Our contribution is to make these
\emph{exact} for the comparative benefit-sign object on the disagreement region: the single
scalar $\gamma$ is shown \emph{necessary} (Theorem~\ref{thm:frontier}(iii)), the frontier
$|M|>\beta$ is shown \emph{tight} with a matching $\tfrac12$ minimax bound
(Proposition~\ref{thm:minimax}), the drift budget is shown \emph{irreducible}
(Corollary~\ref{cor:irreducible}), and the existing label-free estimators are shown to be
its single degenerate face (Proposition~\ref{prop:beta0}). All four statements are validated
numerically (\texttt{val\_frontier.py}; identity exact over $2{\times}10^4$ draws, frontier
law exact over $5{\times}10^5$ draws, ATC error matching the frontier event to machine
precision at drift levels $0.04$--$0.28$).

\subsection{A shared quantity: agreement-on-the-line is the same drift}\label{sec:agl}
Agreement-on-the-Line (\textbf{Miller et al.}\ 2021; \textbf{Baek et al.}\ 2022)---that
out-of-distribution accuracy is predicted label-free by the \emph{agreement} between
models---is proven only for linear/Gaussian families. We show it is governed by the
\emph{same} disagreement-region quantity as the benefit-sign frontier. Fix a reference $f_0$
with $a_0=P(f_0{=}Y)$ and a family $\{f_\theta\}$; on a distribution $P$ write the label-free
agreement $A(\theta)=P(f_\theta{=}f_0)$, the accuracy $a(\theta)=P(f_\theta{=}Y)$, and the
disagreement-region win rate $w(\theta)=P(f_\theta{=}Y\mid f_\theta\neq f_0)$.

\begin{theorem}[Agreement--accuracy law and the shared drift]\label{thm:agl}
\textnormal{(a)} $a(\theta)$ is an exact affine function of the label-free agreement
$A(\theta)$, necessarily through the anchor $(A{=}1,\,a{=}a_0)$, \emph{iff} the win rate
$w(\theta)\equiv\bar w$ is constant across the family; the slope is then $1-2\bar w$.
\textnormal{(b)} Across a source/target pair the two accuracy--agreement lines coincide---so
the unlabeled \emph{agreement} line equals the \emph{accuracy} line---\emph{iff} the win rate
transfers, $\bar w_S=\bar w_T$; the slope gap equals $-2(\bar w_T-\bar w_S)$, exactly twice
the win-rate drift, which is the calibration drift $\gamma$ of \S\ref{sec:bsfrontier} on the
disagreement region.
\textnormal{(c)} Hence non-constant $w$ (spurious correlation, label noise) destroys the line
and nonzero drift tilts the target line off the source line---recovering the documented AGL
failure regimes.
\end{theorem}
\begin{proof}
On $D_\theta=\{f_0\neq f_\theta\}$ (binary $0/1$ loss) exactly one predictor is correct, so by
Lemma~\ref{lem:reduction} $a(\theta)-a_0=\mu(D_\theta)(2w(\theta)-1)=(1-A(\theta))(2w(\theta)-1)$.
If $w\equiv\bar w$ this is affine in $A$ with slope $1-2\bar w$ and value $a_0$ at $A=1$.
Conversely, if $a-a_0=s(A-1)$ for every member then $(1-A)(2w-1)=-s(1-A)$, so $w=(1-s)/2$ is
constant. (b)~The slope on $P$ is $1-2\bar w_P$; equal slopes $\iff\bar w_S=\bar w_T$, and the
gap is $(1-2\bar w_T)-(1-2\bar w_S)=-2(\bar w_T-\bar w_S)$. (c)~immediate.
\end{proof}

\begin{corollary}[One scalar governs both problems]\label{cor:unify-agl}
The benefit-sign frontier (Theorem~\ref{thm:frontier}) and agreement-on-the-line
(Theorem~\ref{thm:agl}) are controlled by the same disagreement-region calibration drift
$\gamma$: \textsc{KGA} certifies adaptation exactly when $|M|>\beta$ (a budget on $\gamma$),
while agreement predicts accuracy exactly when $\gamma=0$. Label-free TTA \emph{gating} and
label-free OOD \emph{accuracy prediction}---posed as separate problems---are two faces of one
quantity.
\end{corollary}

\noindent\emph{Honest scope.} Theorem~\ref{thm:agl} characterizes the \emph{anchored}
accuracy--agreement law (agreement with a fixed reference, raw coordinates); it is
distribution- and model-class-free---strictly more general than the linear/Gaussian proofs of
\textbf{Baek et al.}---but it is \emph{not} the classical probit, random-pairs phenomenon,
whose general characterization remains open. What we establish is the \emph{connection}: the
single scalar $\gamma$ that governs our gate also governs the agreement law and predicts its
failure modes (validated, \texttt{val\_agl.py}: $R^2{=}1.000$ under constant $w$,
$R^2{\approx}0$ under varying $w$, slope $1-2w$ and gap $-2\,\Delta w$ exact).
