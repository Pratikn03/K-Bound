\section{Calibration and held-out evaluation}\label{app:calib-eval}
Algorithm~\ref{alg:calib-eval} gives the evaluation protocol behind the uniform
nine-track panel (Table~\ref{tab:uniform-panel}) and the primary numeric evidence
(Table~\ref{tab:primary-numeric}). Its two branches are \emph{not} interchangeable: stress grids
are scored cell-by-cell with leave-one-cell-out cross-fitting (no single deployment estimator
exists), whereas natural-shift tracks fit one estimator and radius on the development split and
score the untouched held-out domains exactly once.

\begin{algorithm}[t]
\caption{Calibration and evaluation for K-Bound (grid vs.\ natural-shift branches)}
\label{alg:calib-eval}
\begin{algorithmic}[1]
\Require Conditions $\mathcal{C}$; frozen $f_0$; adapters $\mathcal{A}_1,\ldots,\mathcal{A}_K$;
  track type (stress grid or natural shift); level $\alpha$
\Ensure Regret-to-oracle, false-adapt rate, coverage, and action composition
\ForAll{$c \in \mathcal{C}$, adapters $\mathcal{A}_k$} \Comment{shared evidence logging}
    \State $f_a^{(k,c)} \leftarrow \mathcal{A}_k(f_0, X_c)$;\;
      $Z_{k,c} \leftarrow \phi(X_c, f_0, f_a^{(k,c)})$;\;
      $\Delta_{k,c} \leftarrow R_c(f_0) - R_c(f_a^{(k,c)})$
\EndFor
\If{\textbf{stress grid} (CIFAR-10-C, ImageNet-C, PACS): $\mathcal{C}$ is one grid of cells}
  \ForAll{cells $i \in \mathcal{C}$}
    \State Fit $\widehat\Delta_{-i}$ on the other $N{-}1$ cells;\;
      $\rho_i \leftarrow |\widehat\Delta_{-i}(Z_i)-\Delta_i|$ \Comment{leave-one-cell-out}
  \EndFor
  \ForAll{cells $i \in \mathcal{C}$}
    \State $\varepsilon_i \leftarrow \rho_{(k)}$ over $\{\rho_j\}_{j\ne i}$, $k=\lceil N(1-\alpha)\rceil$;\;
      $\varepsilon_i \leftarrow +\infty$ if $k>N{-}1$
      \Comment{\textbf{leave-one-out-of-pool} (\S\ref{par:loo-pool}); unclamped rank rule, so an
      infeasible $k$ returns $+\infty$ and forces \textsc{abstain}. Pool is a jackknife, not a clean
      split: \textbf{NOT exact split conformal}, coverage here is empirical}
  \EndFor
  \ForAll{cells $i \in \mathcal{C}$}
    \State Run Algorithm~\ref{alg:kga} with $(f_0,\mathcal{A}_k,X_i,\widehat\Delta_{-i},\varepsilon_i)$
      \Comment{each cell scored by the estimator that never saw it (cross-fitted)}
  \EndFor
\Else{ \textbf{natural shift} (Camelyon17, iWildCam, Office-Home, RxRx1): split
  $\mathcal{C}=\mathcal{C}_{\mathrm{cal}} \,\dot\cup\, \mathcal{C}_{\mathrm{test}}$ across domains}
  \State Form out-of-fold residuals on $\mathcal{C}_{\mathrm{cal}}$;\;
    $\varepsilon \leftarrow \rho_{(k)}$, $k=\lceil (N{+}1)(1-\alpha)\rceil$ ($+\infty$ if $k>N$)
    \Comment{\textbf{cross-validated residuals applied to a full-data estimator; NOT exact split conformal.} The CV-vs-CV$+$ mismatch means no finite-sample guarantee attaches; jackknife$+$ at level $1-2\alpha$ would, and is not implemented}
  \State Fit deployment estimator $\widehat\Delta(Z)$ on all of $\mathcal{C}_{\mathrm{cal}}$;\;
    freeze $(\widehat\Delta,\varepsilon)$
  \ForAll{$c \in \mathcal{C}_{\mathrm{test}}$}
    \State Run Algorithm~\ref{alg:kga} with $(f_0,\mathcal{A}_k,X_c,\widehat\Delta,\varepsilon)$
      \Comment{untouched held-out domains, scored once}
  \EndFor
\EndIf
\State Record decision, regret-to-oracle, coverage, false-adapt indicator per condition
\State Aggregate metrics vs.\ always-adapt, always-freeze, per-condition oracle
\end{algorithmic}
\end{algorithm}

\paragraph{Asymmetric fixed-policy regret characterization (secondary).}
For completeness, a router strictly improves both fixed-policy regrets only when the optimal
action varies \emph{and} is label-free detectable. The following proposition makes the
condition exact and, crucially, \emph{asymmetric}, which explains why natural-shift tracks tie
(rather than beat) always-adapt. Let $\delta^\star$ be the strict-commitment frontier
(Theorem~\ref{thm:headline}) with abstention resolved to \textsc{freeze}, and let
$R(\pi)=\E_Q[\,|\Delta|\,\mathbf 1\{\pi\neq\text{oracle}\}\,]$ be regret-to-oracle over a
deployment mixture $Q$.
\begin{proposition}[Asymmetric fixed-policy regret]\label{prop:beatsboth-asym}
Assume $\Delta>0$ on $\{M>\beta\}$ and $\Delta<0$ on $\{M<-\beta\}$ for every admissible drift, which
holds by Theorem~\ref{thm:headline}(ii). Then:
\begin{enumerate}
\item[\textnormal{(a)}] $R(\textsc{always-freeze})-R(\delta^\star)
  =\E_Q[|\Delta|\mathbf 1\{M>\beta\}]\ \ge 0$;
\item[\textnormal{(b)}] $R(\textsc{always-adapt})-R(\delta^\star)
  =\E_Q[|\Delta|\mathbf 1\{M<-\beta\}]
  +\E_Q[|\Delta|\mathbf 1\{|M|\le\beta,\ \Delta<0\}]
  -\E_Q[|\Delta|\mathbf 1\{|M|\le\beta,\ \Delta>0\}]$.
\end{enumerate}
Consequently $\delta^\star$ weakly dominates always-freeze, and strictly so if and only if
$\E_Q[|\Delta|\mathbf 1\{M>\beta\}]>0$; whereas $\delta^\star$ beats always-adapt if and only if the
first two terms of (b) exceed the third, i.e.\ the certified plus incidentally-avoided harmful mass
exceeds the helpful adaptation forgone inside the abstention band. Strict improvement over
\emph{both} fixed policies therefore requires detectable-helpful mass and net-detectable-harmful
mass simultaneously.
\end{proposition}
\begin{proof}
Partition the deployment mixture into $\{M>\beta\}$, $\{M<-\beta\}$, and $\{|M|\le\beta\}$. On
$\{M>\beta\}$ the frontier adapts and $\Delta>0$, so $\delta^\star$ is the oracle and pays $0$ while
always-freeze pays $|\Delta|$; on $\{M<-\beta\}$ the frontier freezes and $\Delta<0$, so
$\delta^\star$ pays $0$ while always-adapt pays $|\Delta|$. On $\{|M|\le\beta\}$ the frontier
abstains and, by the stated convention, serves $f_0$: it therefore agrees with always-freeze there
and pays $|\Delta|\mathbf 1\{\Delta>0\}$, while always-adapt pays $|\Delta|\mathbf 1\{\Delta<0\}$.
Taking expectations under $Q$ and differencing term by term gives (a) and (b). Non-negativity in (a)
is immediate; the ``if and only if'' clauses follow because each displayed expectation is an integral
of a non-negative integrand.
\end{proof}
\noindent Because abstention defaults to \textsc{freeze}, the freeze side is free but the
adapt side is conditional: a helpful-dominated shift with negligible detectable-harmful mass
ties or slightly loses to always-adapt. This predicts the candidate-dependent Camelyon17
outcome (Table~\ref{tab:multiseed}): the harmful-mass-bearing arm wins narrowly while the
helpful-dominated arms tie. A $25$-seed numerical check of the partition identity is in the
repository note \texttt{theory\_v2/minimax\_frontier/}. This sharpens, and is consistent with,
the frontier maximality already established in Theorem~\ref{thm:headline}.


\section{Multiclass and regression derivations}\label{app:regression}
For multiclass $0/1$ loss, errors cancel outside $D=\{f_0\neq f_a\}$, giving
$\Delta=\mu_T(D)(p_a-p_0)$ with $p_j=\Pr_T(f_j=Y\mid D)$. In contrast to binary classification,
$p_0$ need not equal $1-p_a$. If $p_a-p_0=M_{\rm mc}+\gamma_{\rm mc}$ and
$|\gamma_{\rm mc}|\le\beta_{\rm mc}$, then $|M_{\rm mc}|>\beta_{\rm mc}$ is sufficient for a
strict commitment; necessity additionally needs the declared class to realize the admissible
evidence-identical drifts.

For squared-loss regression, let $m_T(x)=\E_T[Y\mid X=x]$. Direct expansion gives
\[
\Delta=\E_T\!\left[(f_a-f_0)(2m_T-f_0-f_a)\right].
\]
The integrand vanishes where $f_a=f_0$, so the comparison is again disagreement-local. For an
observable proxy $u(x)$, define
\begin{align*}
M_{\rm reg}&=\E_T[(f_a-f_0)(2u-f_0-f_a)],\\
\gamma_{\rm reg}&=2\E_T[(f_a-f_0)(m_T-u)].
\end{align*}
Then $\Delta=M_{\rm reg}+\gamma_{\rm reg}$. A declared bound
$|\gamma_{\rm reg}|\le\beta_{\rm reg}$ yields the same sufficient strict-commitment rule
$|M_{\rm reg}|>\beta_{\rm reg}$; a converse again requires a richness assumption on the target class.

\section{Auditable drift budgets: statements}\label{app:audit-short}
The drift budget $\beta$ of the population frontier is declared in the main text. The following
results settle when a \emph{computed} budget can replace it. Write $u=\eta_a-s\in[-1,1]$ on $D$ and
$\gamma(P)=\E_{\mu_P}[u\mid D]$. \textbf{Proof status.} Theorem~\ref{thm:short-audA} is proved in
full below; it is the one we rely on. Theorems~\ref{thm:short-audC}, \ref{thm:short-audDE}, and
\ref{thm:short-audG} are \emph{stated here without proof} and are proved in the long version of this
manuscript (appendix ``Auditable drift budgets''); we mark them as such rather than let a reader
assume a refereed proof is available, and no claim in the main text depends on them. Their synthetic
validation is the repository artifact
\texttt{audit\_validation/results\_audited\_drift.json}.

\begin{theorem}[Vacuity of label-free audits]\label{thm:short-audA}
Fix the observables $(\mu_T,f_0,f_a,s)$, hence $M$. Over the class of all target laws with these
observables, any label-free audit $\widehat\beta=g(Z)$ satisfying
$\Pr(|\gamma|\le\widehat\beta)\ge1-\delta$ uniformly must satisfy
$\widehat\beta\ge\tfrac12+|M|$ with probability $\ge1-\delta$; since
$\tfrac12+|M|>|M|$, the audited frontier abstains in every world.
\end{theorem}
\begin{proof}
Because $Z=\phi(X_{1:m},f_0,f_a)$ uses no labels and the observables are held fixed, $\Law(Z)$ --- and
hence the law of $\widehat\beta=g(Z)$ --- is the \emph{same} for every target law in the class; only
the label kernel on $D$ varies. Now $\gamma=\E_{\mu_T}[\eta_a-s\mid D]=\bar a-\tfrac12-M$, and
$\bar a=\Pr_T(f_a{=}Y\mid D)$ ranges over the whole of $[0,1]$ as the label kernel ranges over the
class (take $\eta_a\equiv1$ or $\eta_a\equiv0$ on $D$, both admissible Bernoulli kernels, exactly as
in the construction of Theorem~\ref{lem:nonid}). Hence
$\sup|\gamma|=\max\{|1-\tfrac12-M|,\,|0-\tfrac12-M|\}=\tfrac12+|M|$, attained. Fix any realized value
$b$ of $\widehat\beta$ with $b<\tfrac12+|M|$; choose the extremal kernel, for which
$|\gamma|=\tfrac12+|M|>b$ deterministically, so the coverage event fails on that whole world.
Uniform validity therefore forces $\Pr(\widehat\beta\ge\tfrac12+|M|)\ge1-\delta$.
\end{proof}
\noindent This is the result we regard as carrying the most operational content in the paper, and it
is why $\beta$ is a \emph{declared} input to K-Bound rather than a fitted one: no label-free
procedure can supply a budget that is both valid over the unrestricted class and small enough for the
frontier to ever commit. Every useful audit must therefore purchase structure, which is what
Theorem~\ref{thm:short-audC} prices.

\begin{theorem}[Computed budgets under purchasable structure]\label{thm:short-audC}
Each of the following side informations yields a computable $\widehat\beta$ with
$\Pr(|\gamma(P_T)|\le\widehat\beta)\ge1-\delta$: \textup{(i)} $n$ deployment labels
(Hoeffding; width $\sqrt{2\ln(2/\delta)/n}$); \textup{(ii)} an $L$-Lipschitz drift on a bounded
1-d representation (Kantorovich--Rubinstein $+$ DKW, using only unlabeled batches and labeled
\emph{calibration} data); \textup{(iii)} unknown-direction index drift
$u=g(\theta_*^{\!\top}\psi)$ on $\|\psi\|\le B$ (rotation-uniform adaptive max-sliced audit at rate
$O(\sqrt{(d+\ln(1/\delta))/n})$ via halfspace-VC uniformity; sharper constants from known
max-sliced concentration). More generally, any symmetric witness class of Rademacher/VC
complexity $O(\sqrt{v/n})$ yields a \emph{useful} root-$n$ integral-probability-metric audit
whenever the unknown drift lies in that class (long-manuscript Theorem Aud-H); full Lipschitz
$W_1$ witnesses are excluded from that rate. The structural assumptions are declared and
falsifiable, not verifiable---some such purchase is unavoidable by Theorem~\ref{thm:short-audA}.
\end{theorem}

\begin{theorem}[Composition; fully empirical rule]\label{thm:short-audDE}
For any audit with $\Pr(|\gamma|\le\widehat\beta)\ge1-\delta$: the audited rule (adapt iff
$M>\widehat\beta$, freeze iff $M<-\widehat\beta$, else abstain) falsely strict-commits with
probability $\le\delta$; replacing $M$ by the batch estimate $\widehat M\pm t_M(m,\delta')$
gives a rule computable entirely from data with false-commit probability $\le\delta+\delta'$.
On the synthetic sweep: $0/12$ false commits (empirical rule) and $0/13$ (population rule)
versus $3/20$ for the $\beta{=}0$ plug-in.
\end{theorem}

\begin{theorem}[Domain-level verifiability with an exact floor]\label{thm:short-audG}
With $K$ calibration domains exchangeable with deployment, common per-domain labeled size, and a
drift predictor independent of all $K{+}1$ domains, a domain-level conformal bound certifies
$|\gamma_{K+1}|\le|\gamma_{\mathrm{pred}}(Z_{K+1})|+s_{(j)}+e$ at level $1-\alpha$, feasible iff
$\alpha\ge\tfrac1{K+1}+\delta$. Alignment is thus verifiable exactly at the rate one can afford
exchangeable domains ($K{=}5$: nothing below $\alpha\approx0.17{+}\delta$); per-deployment
verification remains impossible.
\end{theorem}

\noindent\emph{Open (long manuscript).}
\textup{(1)}~\emph{Maximal auditable class:} useful $O(\sqrt{d/n})$ audits exist for
rotation-closed classes whose dual witnesses have root-$n$ complexity (Aud-F/Aud-H), while full
Lipschitz/$W_1$ is rate-excluded; whether anything strictly larger than adaptive MSW$_1$-ridge
admits a useful root-$n$ audit---and whether one-sided audits can beat the spiked-transport
estimation line---remains open (estimation lower bounds alone do not finish the question:
Wasserstein testing can be easier than estimation).
\textup{(2)}~\emph{Certified poly-time supremum:} whether a $\mathrm{poly}(d,n)$ algorithm can
output a sound and null-complete upper bound on linear max-sliced $W_1$ is open; only net
certificates (not gradient-ascent surrogates) are admissible in Aud-F until settled.
