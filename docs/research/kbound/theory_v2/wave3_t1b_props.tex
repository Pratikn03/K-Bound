\begin{proposition}[Exact all-rules lower bound for honest frontier-abstention]
\label{prop:t1b-3world}
Add a third world $\mu=0$ on the frontier and require abstention with probability
$\ge1-\alpha$ in addition to $\mathrm{miss}_\pm\le\alpha$ in worlds $\pm\Delta$. Then every
(possibly randomized) rule needs $n\ge 4\,n_{\mathrm{opt}}$ exactly for every
$\alpha\in(0,\tfrac14)$.
\end{proposition}

\begin{proposition}[Symmetric band achieves $\kappa(\alpha)\,n_{\mathrm{opt}}$, $\kappa\to4$]
\label{prop:t1b-achieve}
In the three-world problem the symmetric band certifies at
$n=\kappa(\alpha)\,n_{\mathrm{opt}}$ with
$\kappa(\alpha)=(1+z_{\alpha/2}/z_\alpha)^2\in[4,\approx5.2]$ for moderate $\alpha$ and
$\kappa(\alpha)\to4$ as $\alpha\to0$.
\end{proposition}
