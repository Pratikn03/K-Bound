\begin{theorem}[Multiclass multicandidate routing with family-wise false-adapt and false-harm control]
\label{thm:multiclass-multicand}
\textbf{Setup (multiclass benefit on the disagreement region).}
Fix base predictor $f_0$ and $K$ candidate adapters $f_a^{(1)},\dots,f_a^{(K)}$.
For $Y\in\{1,\dots,L\}$ with $0/1$ loss, write
$p_0=P_T(f_0{=}Y\mid D)$ and $p_a^{(k)}=P_T(f_a^{(k)}{=}Y\mid D)$.
Off $D$ the per-example benefit is zero; on $D$,
$\E[\delta^{(k)}\mid D]=p_a^{(k)}-p_0$ (Theorem~\ref{thm:disagree-mc}), so
\[
  \Delta_k \;=\; P_T(D)\,\bigl(p_a^{(k)}-p_0\bigr)
  \;=\; \mu_D\,(p_a^{(k)}-p_0),
\]
where $\mu_D:=P_T(D)>0$.  Candidate $k$ is \emph{harmful} if $\Delta_k\le0$ (multiclass
\emph{false-adapt} / \emph{false-harm}: committing to $f_a^{(k)}$ does not reduce $0/1$ risk).

Suppose each candidate admits a label-free lower-confidence bound on its multiclass benefit:
for every $\delta\in(0,1)$,
\begin{equation}\label{eq:mc-percand}
  \Pr\!\bigl[\,\Delta_k < \hat\Delta_k - \varepsilon_k(\,\cdot\,;\delta)\,\bigr]\ \le\ \delta
  \qquad(k=1,\dots,K),
\end{equation}
where $\hat\Delta_k$ is computed from agreement statistics on $D$ without deployment labels
(e.g.\ the plug-in $\hat\mu_D(\hat p_a^{(k)}-\hat p_0)$ of
Theorem~\ref{thm:disagree-mc}).

\textbf{Rule (\textsc{mc-route}$_\alpha$).}
Set $\delta_K=\alpha/K$ (Bonferroni) or $\delta_K=1-(1-\alpha)^{1/K}$ (\v Sid\'ak under
independence).  Form
$S=\{k:\hat\Delta_k-\varepsilon_k(\,\cdot\,;\delta_K)>0\}$ and commit any $\sigma\in S$ if
$S\neq\varnothing$ (e.g.\ $\sigma=\arg\max_{k\in S}(\hat\Delta_k-\varepsilon_k)$), else
\textsc{abstain}.

\textbf{Guarantees.}
\begin{enumerate}
\item[(i)] \emph{Family-wise false-adapt.} With
$\mathrm{FA}^{\mathrm{fw}}:=\{\sigma\in S\ \wedge\ \Delta_\sigma\le0\}$,
$\Pr[\mathrm{FA}^{\mathrm{fw}}]\le\alpha$ under arbitrary dependence among the $K$
per-candidate events (Bonferroni), and under mutual independence (\v Sid\'ak).
\item[(ii)] \emph{Family-wise false-harm (multiclass).} The same bound holds for the
multiclass harm event $\{\sigma\in S\ \wedge\ p_a^{(\sigma)}\le p_0\}$, since
$\Delta_k\le0\iff p_a^{(k)}\le p_0$ when $\mu_D>0$ (Theorem~\ref{thm:disagree-mc}).
\item[(iii)] \emph{Selection-proof.} The bound holds for every data-dependent selector
$\sigma$, including adversarial tie-breaking.
\end{enumerate}
\end{theorem}

\begin{proof}
Part~(i) is Theorem~\ref{thm:multicand} with $\Delta_k$ instantiated as the multiclass benefit
of Theorem~\ref{thm:disagree-mc}; the proof is identical (containment in a union of
per-candidate under-coverage events, then Bonferroni / \v Sid\'ak).  Part~(ii) follows from
$\sign\Delta_k=\sign(p_a^{(k)}-p_0)$ for $\mu_D>0$, so harmfulness in risk equals harmfulness
in accuracy gap on $D$.  Part~(iii) is the selection-proof step of Theorem~\ref{thm:multicand}:
the false-adapt event implies that the committed index's bound under-covers, which is already in
the union over $k$.
\end{proof}

\begin{corollary}[Sample-size penalty for multiclass routing]\label{cor:mc-route-price}
Under sub-Gaussian / empirical-Bernstein radii
$\varepsilon_k(\,\cdot\,;\delta)\asymp\sqrt{2\sigma_k^2\log(1/\delta)/n}$, the Bonferroni
correction enlarges each radius by
$\sqrt{\log(K/\alpha)/\log(1/\alpha)}$: certifying any of $K$ multiclass candidates at
family-wise level $\alpha$ costs $O(\log K)$ in effective sample size, as in
Corollary~\ref{cor:multicand-price}.
\end{corollary}

\begin{remark}[Relation to the open frontier item]\label{rmk:mc-route-scope}
This closes the \emph{routing} half of the ``general/multiclass multicandidate certificates''
target for $0/1$ multiclass loss on $D$: the benefit sign is the ordinal accuracy gap
(Theorem~\ref{thm:disagree-mc}), and family-wise false-adapt / false-harm control is the
Bonferroni lift of Theorem~\ref{thm:multicand}.  What remains open is label-free
\emph{bracketing} of $p_a^{(k)}$ vs.\ $p_0$ without the R1/R2 regularity regime
(Conjecture~\ref{conj:gen}); the certificate logic itself is complete conditional on
\eqref{eq:mc-percand}.
\end{remark}
