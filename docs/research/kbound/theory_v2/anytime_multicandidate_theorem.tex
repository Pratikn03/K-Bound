\begin{assumption}[Parallel multicandidate benefit streams]\label{asm:mc-stream}
Fix $K$ candidates with conditional per-window benefits
$X_t^{(k)}\in[a,b]$ and $\Delta_t^{(k)}:=\E[X_t^{(k)}\mid\mathcal F_{t-1}]$.
Candidate $k$ is \emph{globally harmful} if $\Delta_t^{(k)}\le0$ for all $t$.
No cross-window independence or stationarity is assumed.
\end{assumption}

\begin{theorem}[Anytime family-wise false-adapt for multicandidate routing]
\label{thm:anytime-multicand}
Under Assumption~\ref{asm:mc-stream}, for each $k\in\{1,\dots,K\}$ run the one-sided betting
supermartingale $(E_t^{(k)+})_{t\ge0}$ of Lemma~\ref{lem:smg} (predictable bets, $E_0^{(k)+}=1$).
Fix $\alpha\in(0,1)$ and the \emph{Bonferroni-corrected} anytime rule:
\[
  \textsc{decision}_t=
  \begin{cases}
    \textsc{adapt candidate }k^\star &
      \text{if }\exists\,k:\ E_t^{(k)+}\ge K/\alpha
      \text{ and }k^\star\text{ is any tie-break among such }k,\\[4pt]
    \textsc{abstain} & \text{otherwise,}
  \end{cases}
\]
with committal absorbing at the first crossing.  Then, on the \emph{global harmful null}
\[
  H_0^{\mathrm{fw}}:\ \Delta_t^{(k)}\le0\ \text{for every }k\in\{1,\dots,K\}\text{ and all }t,
\]
\[
  \boxed{\;
  \Pr\!\Bigl(\exists\,t\ge1,\,k:\ \textsc{decision}_t=\textsc{adapt candidate }k\Bigr)
  \;\le\;\alpha\;}
\]
uniformly over all stopping times.  Equivalently: the family-wise anytime false-adapt rate
(committing any harmful candidate at any time) is at most $\alpha$.

\emph{Alternative (e-merging).} Define the arithmetic e-mean
$\bar E_t:=\tfrac1K\sum_{k=1}^K E_t^{(k)+}$ (Vovk--Wang / Ramdas e-merging).  The rule
``adapt candidate $\hat k=\arg\max_k E_t^{(k)+}$ when $\bar E_t\ge K/\alpha$'' satisfies the
same $H_0^{\mathrm{fw}}$ bound by the union bound
$\Pr(\exists k,t:E_t^{(k)+}\ge K/\alpha)\le\sum_k\Pr(\sup_t E_t^{(k)+}\ge K/\alpha)
\le K\cdot(\alpha/K)=\alpha$, using Ville's inequality on each $(E_t^{(k)+})$ under
$\Delta_t^{(k)}\le0$ for all $t$.
\end{theorem}

\begin{proof}
Fix $k$.  Under $H_0^{\mathrm{fw}}$, in particular $\Delta_t^{(k)}\le0$ for all $t$, so
$(E_t^{(k)+})$ is a nonnegative supermartingale with $E_0^{(k)+}=1$
(Lemma~\ref{lem:smg}).  Ville's inequality gives
$\Pr(\sup_{t\ge1}E_t^{(k)+}\ge K/\alpha)\le\alpha/K$.  The family-wise adapt event
\[
  \Bigl\{\exists\,t,k:\ E_t^{(k)+}\ge K/\alpha\Bigr\}
  \;\subseteq\;
  \bigcup_{k=1}^K\Bigl\{\sup_{t\ge1}E_t^{(k)+}\ge K/\alpha\Bigr\}.
\]
By the union bound,
$\Pr(\exists\,t,k:\ \text{adapt }k)\le\sum_{k=1}^K\alpha/K=\alpha$.
Optional stopping follows as in Theorem~\ref{thm:anytime}(i): any stopping time $\tau$ gives
$\{\text{adapt at }\tau\}\subseteq\{\exists t,k:E_t^{(k)+}\ge K/\alpha\}$.
The e-mean remark is the same union bound with a different tie-break; validity does not
require independence across $k$.
\end{proof}

\begin{corollary}[Batch multiclass routing as the $T{=}1$ slice]\label{cor:anytime-mc-batch}
If each $X_1^{(k)}$ is a single-window multiclass benefit estimate with conformal radius
$\varepsilon_k(\alpha/K)$ as in Theorem~\ref{thm:multiclass-multicand}, then the anytime rule
at $t=1$ with $E_1^{(k)+}=\mathbf 1\{\hat\Delta_k-\varepsilon_k(\alpha/K)>0\}/(\alpha/K)$
recovers \textsc{mc-route}$_\alpha$; Theorem~\ref{thm:anytime-multicand} is the time-uniform
extension.  Re-running the batch rule each window without correction inflates the family-wise
rate to $1-(1-\alpha/K)^{KT}\to1$, exactly as in Theorem~\ref{thm:anytime}(iv).
\end{corollary}

\begin{remark}[Multiclass false-harm under anytime routing]\label{rmk:anytime-mc-harm}
On $D$ with $\mu_D>0$, $\Delta_t^{(k)}\le0$ for all $t$ is equivalent to
$p_a^{(k)}\le p_0$ (ordinal accuracy not improved on $D$).  Thus the anytime family-wise bound
controls multiclass \emph{false-harm} on the global harmful null.  Per-window conditional
guarantees after a genuine positive regime (Remark~\ref{rmk:scope}) remain out of scope, as in
the single-candidate anytime theorem.
\end{remark}

\begin{remark}[Numerical validation]\label{rmk:anytime-mc-val}
\texttt{theory\_v2/val\_anytime\_multicandidate.py} checks:
\textbf{[M1]} Bonferroni anytime FWER $\le\alpha$ under $H_0^{\mathrm{fw}}$ for
$K\in\{1,4,16,64\}$;
\textbf{[M2]} naive per-candidate per-window testing inflates above $\alpha$;
\textbf{[M3]} multiclass benefit simulation ($\Delta_k=\mu_D(p_a^{(k)}-p_0)$) matches the
scalar validator of \texttt{val\_multiclass\_multicandidate.py} at $T{=}1$.
\end{remark}
