\section{Calibration and held-out evaluation}\label{app:calib-eval}
Algorithm~\ref{alg:calib-eval} gives the evaluation protocol behind the uniform
nine-track panel (Table~\ref{tab:uniform-panel}), which is the primary numeric
evidence. Its two branches are \emph{not} interchangeable: stress grids
are scored cell-by-cell with leave-one-cell-out cross-fitting (no single deployment estimator
exists), whereas natural-shift tracks fit one estimator and radius on the development split and
score the untouched held-out domains exactly once.

\begin{algorithm}[t]
\caption{Calibration and evaluation for K-Bound (grid vs.\ natural-shift branches)}
\label{alg:calib-eval}
\begin{algorithmic}[1]
\Require Conditions $\mathcal{C}$; frozen $f_0$; adapters $\mathcal{A}_1,\ldots,\mathcal{A}_K$;
  track type (stress grid or natural shift); level $\alpha$
\Ensure Regret-to-oracle, false-adapt rate, coverage, and action composition
\ForAll{$c \in \mathcal{C}$, adapters $\mathcal{A}_k$} \Comment{shared evidence logging}
    \State $f_a^{(k,c)} \leftarrow \mathcal{A}_k(f_0, X_c)$;\;
      $Z_{k,c} \leftarrow \phi(X_c, f_0, f_a^{(k,c)})$;\;
      $\Delta_{k,c} \leftarrow R_c(f_0) - R_c(f_a^{(k,c)})$
\EndFor
\If{\textbf{stress grid} (CIFAR-10-C, ImageNet-C, PACS): $\mathcal{C}$ is one grid of cells}
  \ForAll{cells $i \in \mathcal{C}$}
    \State Fit $\widehat\Delta_{-i}$ on the other $N{-}1$ cells;\;
      $\rho_i \leftarrow |\widehat\Delta_{-i}(Z_i)-\Delta_i|$ \Comment{leave-one-cell-out}
  \EndFor
  \ForAll{cells $i \in \mathcal{C}$}
    \State $\varepsilon_i \leftarrow \rho_{(k)}$ over $\{\rho_j\}_{j\ne i}$, $k=\lceil N(1-\alpha)\rceil$;\;
      $\varepsilon_i \leftarrow +\infty$ if $k>N{-}1$
      \Comment{\textbf{leave-one-out-of-pool} (\S\ref{par:loo-pool}); unclamped rank rule, so an
      infeasible $k$ returns $+\infty$ and forces \textsc{abstain}. Pool is a jackknife, not a clean
      split: \textbf{NOT exact split conformal}, coverage here is empirical}
  \EndFor
  \ForAll{cells $i \in \mathcal{C}$}
    \State Run Algorithm~\ref{alg:kga} with $(f_0,\mathcal{A}_k,X_i,\widehat\Delta_{-i},\varepsilon_i)$
      \Comment{each cell scored by the estimator that never saw it (cross-fitted)}
  \EndFor
\Else{ \textbf{natural shift} (Camelyon17, iWildCam, Office-Home, RxRx1): split
  $\mathcal{C}=\mathcal{C}_{\mathrm{cal}} \,\dot\cup\, \mathcal{C}_{\mathrm{test}}$ across domains}
  \State Form out-of-fold residuals on $\mathcal{C}_{\mathrm{cal}}$;\;
    $\varepsilon \leftarrow \rho_{(k)}$, $k=\lceil (N{+}1)(1-\alpha)\rceil$ ($+\infty$ if $k>N$)
    \Comment{\textbf{cross-validated residuals applied to a full-data estimator; NOT exact split conformal.} The CV-vs-CV$+$ mismatch means no finite-sample guarantee attaches; jackknife$+$ at level $1-2\alpha$ would, and is not implemented}
  \State Fit deployment estimator $\widehat\Delta(Z)$ on all of $\mathcal{C}_{\mathrm{cal}}$;\;
    freeze $(\widehat\Delta,\varepsilon)$
  \ForAll{$c \in \mathcal{C}_{\mathrm{test}}$}
    \State Run Algorithm~\ref{alg:kga} with $(f_0,\mathcal{A}_k,X_c,\widehat\Delta,\varepsilon)$
      \Comment{untouched held-out domains, scored once}
  \EndFor
\EndIf
\State Record decision, regret-to-oracle, coverage, false-adapt indicator per condition
\State Aggregate metrics vs.\ always-adapt, always-freeze, per-condition oracle
\end{algorithmic}
\end{algorithm}

\section{Multiclass and regression derivations}\label{app:regression}
For multiclass $0/1$ loss, errors cancel outside $D=\{f_0\neq f_a\}$, giving
$\Delta=\mu_T(D)(p_a-p_0)$ with $p_j=\Pr_T(f_j=Y\mid D)$. In contrast to binary classification,
$p_0$ need not equal $1-p_a$. If $p_a-p_0=M_{\rm mc}+\gamma_{\rm mc}$ and
$|\gamma_{\rm mc}|\le\beta_{\rm mc}$, then $|M_{\rm mc}|>\beta_{\rm mc}$ is sufficient for a
strict commitment; necessity additionally needs the declared class to realize the admissible
evidence-identical drifts.

For squared-loss regression, let $m_T(x)=\E_T[Y\mid X=x]$. Direct expansion gives
\[
\Delta=\E_T\!\left[(f_a-f_0)(2m_T-f_0-f_a)\right].
\]
The integrand vanishes where $f_a=f_0$, so the comparison is again disagreement-local. For an
observable proxy $u(x)$, define
\begin{align*}
M_{\rm reg}&=\E_T[(f_a-f_0)(2u-f_0-f_a)],\\
\gamma_{\rm reg}&=2\E_T[(f_a-f_0)(m_T-u)].
\end{align*}
Then $\Delta=M_{\rm reg}+\gamma_{\rm reg}$. A declared bound
$|\gamma_{\rm reg}|\le\beta_{\rm reg}$ yields the same sufficient strict-commitment rule
$|M_{\rm reg}|>\beta_{\rm reg}$; a converse again requires a richness assumption on the target class.

\subsection{Multiclass and Regression Scope}
The binary $0$--$1$ loss case gives the cleanest frontier because, on $D=\{x:f_0(x)\neq f_a(x)\}$, exactly one predictor is correct. For multiclass $0$--$1$ loss,
\[
\Delta=P_T(D)(p_a-p_0),\qquad p_j=P_T(f_j(X)=Y\mid D),
\]
so the decision becomes an ordinal comparison of the two disagreement-region accuracies. For squared-loss regression, the risk difference reduces to a disagreement-local moment. The main theorem uses a split-observable decomposition $\sign(\Delta)=\sign(M+\gamma)$; the multiclass bridge is stated below and the regression derivation appears in Appendix~\ref{app:regression}.

\begin{proposition}[Multiclass benefit decomposition]\label{prop:multiclass}
For multiclass $0/1$ loss, $\Delta=\mu_T(D)(p_a-p_0)$ with $p_j=\Pr_T(f_j(X){=}Y\mid D)$. If the
disagreement-region accuracy gap admits a split-observable decomposition
$p_a-p_0=M_{\mathrm{mc}}+\gamma_{\mathrm{mc}}$ with $|\gamma_{\mathrm{mc}}|\le\beta_{\mathrm{mc}}$, then
$|M_{\mathrm{mc}}|>\beta_{\mathrm{mc}}$ implies $\sign\Delta=\sign M_{\mathrm{mc}}$. The converse
strict-commitment result additionally requires a richness assumption: the declared class must contain
augmented-evidence-identical laws whose admissible $\gamma_{\mathrm{mc}}$ values straddle
$-M_{\mathrm{mc}}$, together with the zero-benefit boundary law. At
$M_{\mathrm{mc}}=\beta_{\mathrm{mc}}=0$, the sign is identified as zero.
\end{proposition}
\begin{proof}
Off $D$ the two predictors agree, so their $0/1$ losses cancel pointwise and only $D$ contributes:
$\Delta=\mu_T(D)\bigl(\Pr_T(f_a{=}Y\mid D)-\Pr_T(f_0{=}Y\mid D)\bigr)=\mu_T(D)(p_a-p_0)$. Since
$\mu_T(D)>0$, $\sign\Delta=\sign(p_a-p_0)=\sign(M_{\mathrm{mc}}+\gamma_{\mathrm{mc}})$; if
$M_{\mathrm{mc}}>\beta_{\mathrm{mc}}\ge|\gamma_{\mathrm{mc}}|$ the sum is positive, and symmetrically
for the negative case. The converse repeats the label-kernel construction of
Theorem~\ref{lem:nonid} with the multiclass conditional, which requires the declared class to
contain the resulting evidence-identical laws --- hence the richness hypothesis. Full derivation in
Appendix~\ref{app:regression}.
\end{proof}
\noindent In the multiclass experiments \textsc{KGA} estimates $\Delta$ directly through the benefit
proxy $\widehat\Delta=h_\theta(Z)$ (not $M_{\mathrm{mc}}$ itself); the binary frontier is the
identifiability backbone, and the multiclass results are evaluated through this benefit-estimation route.

backward computation before commitment. The controller's \emph{added} cost is measured by a

\section{Auditable drift budgets: the positive side}\label{app:audit-short}\label{app:auditable}
The impossibility half of the budget question is Theorem~\ref{thm:beta-minimax} and
Corollary~\ref{cor:audA} in the main text. This appendix collects the \emph{positive} side: the
structural purchases under which a budget can be computed, and the rate each one achieves. Their
role after \S\ref{sec:impossible} is precise: \textbf{each is an instance of
Proposition~\ref{prop:threeterm}. Each purchases the first term, $|\gamma_{\mathrm{cal}}|$, with
labels at an anchor; each sets the third term $\rho$ to zero by a declared invariance; each
estimates the middle term in a class-specific way; and none of them omits the first term.} That last
fact is the non-vacuity certificate for \S\ref{sec:impossible}: if some audit here computed a budget
with no labels anywhere, Theorem~\ref{thm:beta-minimax} would be false. Write $u=\eta_a-s\in[-1,1]$ on $D$ and $\gamma(P)=\E_{\mu_P}[u\mid D]$.

\begin{theorem}[Computed budgets under purchasable structure]\label{thm:short-audC}\label{thm:aud-C}
Assume \textup{(Lip)}: $u$ is $L$-Lipschitz on $D$ w.r.t.\ a bounded 1-d representation
($|\psi|\le B$). With $n_{\mathrm{lab}}$ labeled calibration points, $m_{\mathrm{cal}},
m_{\mathrm{dep}}$ unlabeled points and empirical measures $\widehat\mu$, let
$\varepsilon_{\mathrm{DKW}}(m,\delta)=\sqrt{\ln(2/\delta)/(2m)}$ and define
\begin{align*}
\widehat\beta_{W}
&=|\widehat\gamma_{\mathrm{cal}}|+t(n_{\mathrm{lab}},\delta/2)\\
&\quad+L\widehat W_1(\widehat\mu_T,\widehat\mu_{\mathrm{cal}})\\
&\quad+2BL\varepsilon_{\mathrm{DKW}}(m_{\mathrm{cal}},\delta/4)\\
&\quad+2BL\varepsilon_{\mathrm{DKW}}(m_{\mathrm{dep}},\delta/4).
\end{align*}
We have $\Pr(|\gamma(P_T)|\le\widehat\beta_{W})\ge1-\delta$.
More generally, any symmetric witness class of Rademacher/VC complexity $O(\sqrt{v/n})$ yields a
\emph{useful} root-$n$ integral-probability-metric audit whenever the unknown drift lies in that
class; full Lipschitz $W_1$ witnesses are excluded from that rate. The structural assumptions are
declared and falsifiable, not verifiable---some such purchase is unavoidable by
Corollary~\ref{cor:audA}.
\end{theorem}
\begin{proof}
\emph{Population step:} $u/L$ is 1-Lipschitz, so Kantorovich--Rubinstein gives
$|\gamma(P_T)-\gamma(P_{\mathrm{cal}})|\le L\,W_1(\mu_T,\mu_{\mathrm{cal}})$.
\emph{Finite-sample:} on a 1-d bounded representation,
$W_1(\widehat\mu,\mu)=\int|\widehat F-F|\le 2B\sup_t|\widehat F-F|$,
bounded by DKW at level $\delta/4$ per measure; Hoeffding at $\delta/2$ for
$\widehat\gamma_{\mathrm{cal}}$; union bound and the $W_1$ triangle inequality assemble
$\widehat\beta_W$. \textup{(Lip)} is declared, not verified---Theorem~\ref{thm:beta-minimax}
guarantees some such purchase is unavoidable; multi-domain Lipschitz-ratio checks falsify but
cannot verify it. Their synthetic validation is the repository artifact
\texttt{audit\_validation/results\_audited\_drift.json}.
\end{proof}

\begin{theorem}[Composition; fully empirical rule]\label{thm:short-audDE}\label{thm:aud-DE}
Let $\widehat\beta$ be any audit with $\Pr(|\gamma|\le\widehat\beta)\ge1-\delta$.
\textup{(D)}~The audited population rule (adapt iff $M>\widehat\beta$; freeze iff
$M<-\widehat\beta$; else abstain) has $\Pr(\text{false strict commitment})\le\delta$.
\textup{(E)}~With $\widehat M=\tfrac1m\sum s(X_i)-\tfrac12$ from $m$ unlabeled deployment
points and $t_M(m,\delta')=\sqrt{\ln(2/\delta')/(2m)}$, the \emph{empirical} audited rule
(adapt iff $\widehat M-t_M>\widehat\beta$; freeze iff $\widehat M+t_M<-\widehat\beta$) has
$\Pr(\text{false strict commitment})\le\delta+\delta'$, and every quantity in the rule is
computable. On the synthetic sweep: $0/12$ false commits (empirical rule) and $0/13$
(population rule) versus $3/20$ for the $\beta{=}0$ plug-in.
\end{theorem}
\begin{proof}
\textup{(D)}: on $\{|\gamma|\le\widehat\beta\}$, $M>\widehat\beta\ge|\gamma|$ implies
$M+\gamma>0$, so $\sign\Delta=+1$ by Lemma~\ref{lem:reduction}; symmetric for freeze; a false
commitment requires the audit event to fail.
\textup{(E)}: Hoeffding for the $[0,1]$-valued $s(X_i)$ gives
$\Pr(|\widehat M-M|>t_M)\le\delta'$; on the intersection of the two events the adapt branch gives
$M\ge\widehat M-t_M>\widehat\beta\ge|\gamma|$; union bound.
\end{proof}

\begin{theorem}[Domain-level verifiability with an exact floor]\label{thm:short-audG}\label{thm:aud-G}
Assume \textup{(G1)} the $K$ calibration domains and the deployment domain are exchangeable
tuples $(P_k,Z_k,\gamma_k)$; \textup{(G2)} each calibration domain carries a labeled sample of
\emph{common} size $n_{\mathrm{lab}}$, giving $\widehat\gamma_k$ with common Hoeffding radius $e$
at level $\delta$; \textup{(G3)} the label-free drift predictor $\gamma_{\mathrm{pred}}(\cdot)$
is fixed and independent of all $K{+}1$ domains. Let
$s_k=|\widehat\gamma_k-\gamma_{\mathrm{pred}}(Z_k)|$ and
$j=\lceil(1-\alpha+\delta)(K+1)\rceil$, feasible iff $\alpha\ge\tfrac1{K+1}+\delta$. Then
\[
\Pr\Bigl(\,|\gamma_{K+1}|\le|\gamma_{\mathrm{pred}}(Z_{K+1})|+s_{(j)}+e\,\Bigr)\ \ge\ 1-\alpha,
\]
with every right-hand quantity computable without deployment labels. Alignment is thus verifiable
exactly at the rate one can afford exchangeable domains ($K{=}5$: nothing below
$\alpha\approx0.17{+}\delta$); per-deployment verification remains impossible.
\end{theorem}
\begin{proof}
The computable scores $s_1,\dots,s_{K+1}$ are exchangeable under (G1)--(G3) (if per-domain
labeled sizes differ, exchangeability fails and validity is lost). Split conformal at level
$\alpha'=\alpha-\delta$ gives $\Pr(s_{K+1}\le s_{(j)})\ge1-\alpha'$. One deployment-side
Hoeffding event bounds the ideal score
$S_{K+1}=|\gamma_{K+1}-\gamma_{\mathrm{pred}}(Z_{K+1})|\le s_{K+1}+e$; the deterministic
triangle $|\gamma_{K+1}|\le|\gamma_{\mathrm{pred}}(Z_{K+1})|+S_{K+1}$ and a union bound finish.
\emph{Floor:} the conformal index requires $j\le K$, i.e.\ $\alpha\ge\tfrac1{K+1}+\delta$
exactly; per-deployment verification remains impossible (Theorem~\ref{thm:beta-minimax});
verifiability is \emph{priced in exchangeable domains}. (G3) is essential: a predictor fit on
the calibration domains breaks exchangeability.
\end{proof}
