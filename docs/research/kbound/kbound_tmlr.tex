%% kbound_tmlr.tex -- maintained single-column long-form K-Bound companion.
%%
%% This driver uses the same corrected abstract, 11-section body, generated
%% numbers, theorem sources, appendices, and bibliography as the compact paper.
%% The source-hashed canonical panel remains the numerical authority for both
%% layouts. Build this companion with BUILD_LONG_TMLR=1.
%%
%% WHY THIS FILE EXISTS
%%   The IEEEtran two-column build (kbound_short.tex) ran to 23 pages with 26
%%   tables, which is a desk-reject risk at a conference and, more importantly,
%%   is too dense for a reviewer to hold in mind. TMLR is single-column with no
%%   hard page limit and judges claim-evidence support rather than novelty
%%   magnitude, which is the right bar for this work. Reformatting single-column
%%   dissolves the length problem WITHOUT cutting any result table.
%%
%% >>> BEFORE SUBMITTING: DROP tmlr.sty NEXT TO THIS FILE <<<
%%   The official TMLR style file (tmlr.sty, from
%%   https://github.com/JmlrOrg/tmlr-style-file) is NOT vendored here, because
%%   redistributing it inside this repository is not ours to do. Download it,
%%   place it in this directory, and swap in these TWO lines:
%%
%%     (1) replace   \documentclass[11pt]{article}
%%         with      \documentclass{article}\usepackage{tmlr}
%%                   % ... or \usepackage[accepted]{tmlr} for the camera-ready,
%%                   %     or \usepackage[preprint]{tmlr} for an arXiv preprint.
%%
%%     (2) delete the "TMLR SHIM" block below (between the two marker comments)
%%         -- tmlr.sty defines \title/\author/\abstract handling, the running
%%         head, and the anonymisation switch itself, and the shim will clash.
%%
%%   Everything else in this file, and every file it \input s, is style-agnostic.
%%
%% ANONYMITY
%%   TMLR is single-blind by default: author names are shown. This file therefore
%%   compiles NON-anonymous. Set \anontrue below (or \usepackage{tmlr} without
%%   [accepted]) for a double-blind venue; \ifanon also suppresses the repository
%%   link in the Reproducibility section, which is a real de-anonymisation hazard
%%   (the released tree contains CITATION.cff with the author's name and email).
%%
%% SHARED SOURCE
%%   kbound_abstract.tex          shared abstract
%%   kbound_submission_body.tex  shared main body and vetted appendices
%%   paper/references_kbound_expanded.tex
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%% Delete this whole block once tmlr.sty is present (see header, step 2).
%% Geometry approximating tmlr.sty's text block (6.5in x 9in on US letter) so that
%% the page count you see here is close to the page count you will get with the real
%% style file. Do NOT load parskip: tmlr.sty uses indented paragraphs with no inter-
%% paragraph glue, and parskip inflates this document by roughly a fifth.
\usepackage[letterpaper,margin=1in]{geometry}
% Spacing only, no content change.  Mirrors the setting the companion driver
% (kbound.tex) already carries.  Absorbs the residual overfull hboxes caused by
% long unbreakable \texttt{} paths in the appendices.
\setlength{\emergencystretch}{3em}
%% \path is IEEEtran/url sugar used throughout the shared body.
\providecommand{\path}[1]{\texttt{#1}}
%% article already defines table* / figure* (they degrade to plain floats in

%% shared body's floats need no aliasing.
%% IEEEtran-only front-matter commands used by the shared body / the other driver:
\providecommand{\IEEEauthorblockN}[1]{{\large #1}}
\providecommand{\IEEEauthorblockA}[1]{{\normalsize #1}}
\newenvironment{IEEEkeywords}{\par\noindent\textbf{Keywords:}\ }{\par}
%% IEEEtran's \appendices -> article's \appendix, with lettered sections.
\providecommand{\appendices}{\appendix}
