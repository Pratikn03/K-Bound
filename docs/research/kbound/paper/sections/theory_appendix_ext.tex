\begin{lemma}[Plug-in regret identity]\label{lem:gate}
For any estimator $\hat\Delta$ and gate $\hat g=\mathbf 1[\hat\Delta>0]$,
\[
R(\hat g)-R(g^\star)=\E\bigl[|\Delta(Z)|\,\mathbf 1\{\sign\hat\Delta\neq\sign\Delta\}\bigr],
\qquad g^\star=\mathbf 1[\Delta>0].
\]
On $\{|\Delta|<\varepsilon\}$, committal regret is at most $\varepsilon$.
\end{lemma}
\begin{proof}
Wrong-sign commits pay $|\Delta|$ pointwise; averaging gives the identity.
\end{proof}

\begin{proposition}[Finite-sample minimax regret floor]\label{prop:lecam-finite}
Let two target worlds share evidence laws at TV distance $\delta$ and have opposite benefit
signs with magnitude $\Lambda>0$. For every label-free committal gate and finite $n$,
\[
\inf_{\hat g}\ \max_{\theta=\pm1}\bigl[R_{P_\theta}(\hat g)-R_{P_\theta}(g^\star)\bigr]
\ \ge\ \tfrac{\Lambda}{2}(1-\delta)
\ \ge\ \tfrac{\Lambda}{4}\,e^{-\mathrm{KL}}.
\]
A Gaussian family with $d_n=c/\sqrt n$ realizes a constant floor in $n$
(\texttt{val\_thm2\_lecam\_finite\_n.py}).
\end{proposition}

\paragraph{Total variation.} For probability measures $P,Q$ on a common measurable space,
$\TV(P,Q):=\sup_A|P(A)-Q(A)|\in[0,1]$; equivalently $\TV=\tfrac12\|P-Q\|_1$ when both are dominated
by a common measure.

\begin{theorem}[Matched evidence impossibility (full form)]\label{thm:imp}
Theorem~\ref{lem:nonid} establishes part~(i) of the following (only part (i); (ii) and (iii) are
proved here). Fix $\alpha\in(0,\tfrac12)$.
\begin{enumerate}
\item[\textnormal{(i)}] Witness worlds with $\Law(Z\mid P_T^1)=\Law(Z\mid P_T^2)$ and
$\sign\Delta_1=-\sign\Delta_2\neq0$.
\item[\textnormal{(ii)}] For any test $g$ deciding between the two worlds from $Z$, writing
$\mathrm{err}(g):=\Pr_{1}[g{=}2]+\Pr_{2}[g{=}1]$ for the \emph{sum} of the two error
probabilities (not their average, and unrelated to the observable margin $M$),
$\inf_g\mathrm{err}(g)=1-\TV\bigl(\Law(Z\mid P_T^1),\Law(Z\mid P_T^2)\bigr)$. On the witness pair of
(i) the two laws coincide, so $\TV=0$ and $\inf_g\mathrm{err}(g)=1$.
\item[\textnormal{(iii)}] Forced abstention at rate $\ge 1-2\alpha$ under dual error control.
\end{enumerate}
\end{theorem}
\begin{proof}
(i) is Theorem~\ref{lem:nonid}. For (ii), let $\mu$ dominate both laws with densities $p_1,p_2$.
For any (possibly randomized) test with rejection function $\psi\in[0,1]$,
$\mathrm{err}=\int\psi p_1+\int(1-\psi)p_2 = 1-\int\psi(p_2-p_1)$, which is minimized pointwise by
the Neyman--Pearson choice $\psi=\mathbf 1\{p_2>p_1\}$, giving
$\mathrm{err}=1-\int(p_2-p_1)^+=1-\TV$. Randomization on the tie set does not change the value.
For (iii), a label-free rule sees only $\Law(Z)$, which is the same in both worlds, so its action
probabilities $(q_a,q_f,q_{\mathrm{ab}})$ are identical in both. Dual error control forces
$q_a\le\alpha$ in the world with $\Delta<0$ and $q_f\le\alpha$ in the world with $\Delta>0$; since
the probabilities are world-independent and sum to one, $q_{\mathrm{ab}}=1-q_a-q_f\ge1-2\alpha$.
\end{proof}

\begin{corollary}[Closed-band abstention under dual error control]\label{cor:forced-abstain}
In the $|M|\le\beta$ wedge of Theorem~\ref{thm:headline}, Theorem~\ref{thm:imp}(iii) implies
any label-free rule with false-adapt and false-freeze $\le\alpha$ must abstain with probability at
least $1-2\alpha$ on matched evidence. Opposite nonzero signs are required only for $|M|<\beta$;
at $|M|=\beta>0$ zero-versus-strict ambiguity suffices, and at $M=\beta=0$ the benefit is
identically zero so neither strict action is sound.
\end{corollary}
\begin{proof}
Immediate from Theorem~\ref{thm:imp}(iii) applied to the witness pair supplied by
Theorem~\ref{lem:nonid} when $|M|<\beta$, and to the zero-benefit boundary law of
Proposition~\ref{prop:closed-band} when $|M|=\beta>0$ (there the ``error'' event in the zero world is
$\{\Delta\le0\}\cap\{\Delta\ge0\}$, which both strict actions incur).
\end{proof}

\begin{remark}[$\beta=0$ face]\label{rem:beta-zero}
On the $\beta=0$ face the class $\mathcal{C}_0$ forces $\gamma=0$, so
Lemma~\ref{lem:reduction} gives $\sign\Delta=\sign M$ exactly, and the frontier of
Theorem~\ref{thm:headline} specializes as follows. If $M\neq0$, the strict action of sign $M$
(\textsc{adapt} for $M>0$, \textsc{freeze} for $M<0$) is uniformly sound over $\mathcal{C}_0$.
If $M=0$, then $\Delta=0$ identically, so under the strict-decision semantics neither strict
action is sound and the maximal sound action is \textsc{abstain}. Existing label-free estimators
may be interpreted as $\beta=0$ plug-ins only when their estimand realizes the stated
disagreement-score decomposition; assuming $\gamma=0$ alone does not establish their soundness.
\end{remark}

\begin{theorem}[One-bit dichotomy]\label{thm:conj1-dichotomy}
Under the CEI panel hypothesis \textbf{H}: \textnormal{(i)}~rank-one agreement law;
\textnormal{(ii)}~evidence fixes magnitudes but not global orientation $b\mapsto-b$;
\textnormal{(iii)}~no evidence-definable assumption identifies $\sign\Delta$; the minimal
untestable supplement is one bit; \textnormal{(iv)}~drift budgets are auditable but not
verifiable on a blind set. Conjecture~\ref{conj:gen} is resolved negatively by~(iii).
\end{theorem}

\begin{conjecture}[Label-free bracketing --- \textbf{resolved}]\label{conj:gen}
Unconditional label-free bracketing does not exist; the weakest one-bit classes (an explicit finite
family of dominance polytopes) are characterized in the long manuscript (\texttt{kbound.tex}).
\end{conjecture}

\begin{theorem}[Evidence-channel rate]\label{thm:ev-rate}
Under audited H, unlabeled agreement statistics and labeled paired benefits both converge at
rate $m^{-1/2}$; labels change constants and supply the missing bit, not the exponent.
\end{theorem}
