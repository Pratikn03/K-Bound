The body is organized as four pillars, with five core theorems serving as the load-bearing
statements inside that scaffold. \textbf{Pillar I (impossibility floor):}
Theorem~\ref{thm:imp} plus the Le Cam lower bound and regret identity establish when
abstention is information-theoretically forced. \textbf{Pillar II (certificate + audit):}
Theorem~\ref{thm:cert} gives finite-sample false-adapt control, and the one-bit audit
machinery of Theorem~\ref{thm:conj1-dichotomy}(iv) supplies a common falsifier for the
legacy budget premises (formerly presented as separate propositions, now treated as
corollaries of one audit framework). \textbf{Pillar III (identifiable regimes):}
Theorem~\ref{thm:frontier} and Theorem~\ref{thm:conj1-dichotomy} characterize exactly what
label-free evidence identifies and what one-bit supplement remains irreducible.
\textbf{Pillar IV (rates):} Theorem~\ref{thm:ev-rate} gives matching evidence-channel rates.
The open piece---a weakest falsifiable class on which one declared bit suffices---is settled
in Appendix~\ref{app:weakest}: conditionally for the single class $\mathcal C_{\mathrm{mono}}$, and
unconditionally as an explicit finite family of dominance polytopes (Theorem~\ref{thm:uncond-weakest}). Supporting variants (multiclass/regression
reductions, label-shift and drift boundaries, ambiguity-reach unification, router capacity,
anytime certificates, the $\gamma$-meter, and the agreement--accuracy law) are presented as
corollaries or companions in the companion manuscript, each with a validator; every numeric
claim here traces to a script in the public repository.

\begin{lemma}[Reduction to the disagreement region]\label{lem:reduction}\label{thm:disagree}
Let $D=\{x:f_0(x)\neq f_a(x)\}$ (observable). For binary $Y$ and $0/1$ loss, with
$\bar a=\Pr_T(f_a{=}Y\mid D)$,
$\Delta=2\,\mu(D)(\bar a-\tfrac12)$, so for $\mu(D)>0$:
$\sign\Delta=\sign(\bar a-\tfrac12)$. Writing $s$ for any source-calibrated score of
$f_a$, $M:=\E_{\mu\mid D}[s]-\tfrac12$ (observable) and
$\gamma:=\E_{\mu\mid D}[\eta_a-s]$ (unobservable),
$\boxed{\,\sign\Delta=\sign(M+\gamma)\,}$.
For $K$-class $0/1$ loss $\Delta=\mu(D)(p_a-p_0)$, and for squared loss the sign reduces
to one moment on $D$; both extend the identity verbatim (manuscript, App.~C).
\end{lemma}
\begin{proof}[Proof sketch]
Off $D$ the losses coincide; on $D$ exactly one of $f_0,f_a$ is correct (binary), giving
$\E[\delta\mid D]=2\bar a-1$; linearity splits $\bar a-\tfrac12=M+\gamma$. Validated on synthetic draws (identity exact to machine precision; multiclass to
$10^{-16}$ over $4000$ trials).
\end{proof}

\begin{lemma}[Plug-in regret identity]\label{thm:gate}
For any estimator $\hat\Delta$, the gate $\hat g=\mathbf 1[\hat\Delta>0]$ satisfies
$R(\hat g)-R(g^\star)=\E\big[|\Delta(Z)|\,\mathbf 1\{\sign\hat\Delta\neq\sign\Delta\}\big]$
with $g^\star=\mathbf 1[\Delta>0]$ Bayes; on $\{|\Delta|<\varepsilon\}$ committal regret
is $\le\varepsilon$, justifying abstention in the low-margin band. Validated to machine precision on synthetic regret identities.
\end{lemma}

\begin{proposition}[Finite-sample minimax regret floor for label-free gating]\label{prop:lecam-finite}\label{cor:lecam-regret-floor}
\emph{Intuition.} The identity of Lemma~\ref{thm:gate} is an equality, not a hardness
statement; pairing it with Le Cam's two-point method turns it into a finite-$n$ lower bound
on the regret of \emph{any} label-free gate, which does not vanish as $n\to\infty$ when the
two worlds are rescaled to stay equally hard.
Let $\{P_\theta\}_{\theta\in\{-1,+1\}}$ be two target worlds sharing the same disagreement
structure with constant benefit magnitude $|\Delta(P_\theta)|\equiv\Lambda>0$ and opposite
signs $\sign\Delta(P_\theta)=\theta$, and let $Q^{(n)}_\theta$ denote the law of \emph{any}
label-free evidence $Z$ computed from $n$ unlabeled observations under $P_\theta$. Then for
every label-free committal gate $\hat g:Z\mapsto\{\textsc{adapt},\textsc{freeze}\}$ and every
finite $n$,
\begin{multline*}
\inf_{\hat g}\ \max_{\theta\in\{\pm1\}}\big[R_{P_\theta}(\hat g)-R_{P_\theta}(g^\star)\big]\\
\ge\ \tfrac{\Lambda}{2}\Big(1-\TV\big(Q^{(n)}_{-1},Q^{(n)}_{+1}\big)\Big)\\
\ge\ \tfrac{\Lambda}{4}\,e^{-\mathrm{KL}(Q^{(n)}_{-1}\|Q^{(n)}_{+1})}.
\end{multline*}
In particular there is an explicit Gaussian two-point family ($X\sim\mathcal N(\theta d_n,1)$ on
$D$, $f_0=\mathbf1[x>0]$, $f_a=\mathbf1[x<0]$, with the target label rule giving $\Delta=\theta$,
so $\Lambda=1$) for which, taking $d_n=c/\sqrt n$, both bounds are \emph{constant in $n$}:
$\TV(Q^{(n)}_{-1},Q^{(n)}_{+1})=2\Phi(c)-1$ and the floor equals $\Lambda\,\Phi(-c)$, with the
Bretagnolle--Huber certificate $\tfrac{\Lambda}{4}e^{-2c^2}>0$. Theorem~\ref{thm:imp} is the
$c\to0$ (TV${}\to0$) face, where the floor saturates at $\Lambda/2$.
\end{proposition}
\begin{proof}
By Lemma~\ref{thm:gate}, since $|\Delta|\equiv\Lambda$ and $g^\star$ is the constant correct
action $\sign\theta$ in world $\theta$,
$R_{P_\theta}(\hat g)-R_{P_\theta}(g^\star)=\Lambda\,Q^{(n)}_\theta(\hat g\text{ commits to the
wrong sign})$. Averaging over the uniform prior on $\theta$ gives
$\tfrac12\sum_\theta\big[R_{P_\theta}(\hat g)-R_{P_\theta}(g^\star)\big]
=\tfrac{\Lambda}{2}M(\hat g)$, where
$M(\hat g)=Q^{(n)}_{+1}(\hat g{=}\textsc{freeze})+Q^{(n)}_{-1}(\hat g{=}\textsc{adapt})$ is the
mixed committal error of Theorem~\ref{thm:imp}(ii). Le Cam's testing-affinity identity yields
$\inf_{\hat g}M(\hat g)=1-\TV(Q^{(n)}_{-1},Q^{(n)}_{+1})$, and $\max\ge\mathrm{avg}$ gives the
minimax form; Bretagnolle--Huber ($1-\TV\ge\tfrac12 e^{-\mathrm{KL}}$) gives the closed-form
certificate. For the Gaussian family the sufficient statistic $\bar X\sim\mathcal N(\theta
d_n,1/n)$ has $\TV=2\Phi(\sqrt n\,d_n)-1$ and per-observation
$\mathrm{KL}=2d_n^2$, so $d_n=c/\sqrt n$ makes both $n$-independent; $|\Delta|=1$ exactly because
each label rule makes one of $f_0,f_a$ correct on a probability-one event.
Validated (\texttt{val\_thm2\_lecam\_finite\_n.py}, seeds fixed): across $c\in\{0.25,0.5,1,1.5\}$
and $n\in\{10,50,200,1000\}$ the empirical floor is constant in $n$, the Bayes (sample-mean
threshold) gate \emph{meets} it (so it is tight), and a brute-force minimax over a panel of
label-free rules (all thresholds/polarities on five statistics) never beats it within
Monte-Carlo error.
\end{proof}

\begin{theorem}[The unknowable regime is real and tight]\label{thm:imp}\label{thm:imp-quant}\label{cor:forced-abstain}
\emph{Intuition.} Identical label-free evidence can support opposite adaptation benefits; any
valid rule must then abstain rather than guess.
\textnormal{(i)} There exist target worlds $P_T^1,P_T^2$ with identical evidence laws,
$\Law(Z\mid P_T^1)=\Law(Z\mid P_T^2)$, and $\sign\Delta_1=-\sign\Delta_2\neq0$
(explicit witness: $X\sim\mathcal N(0,1)$, $f_a=1-f_0$, flipped labels). No label-free
rule distinguishes them; the worst-case-optimal committal-regret action is
\textsc{abstain}.
\textnormal{(ii)} Quantitatively, for any committal rule $g$ the mixed error obeys the
Le Cam identity $\inf_g M(g)=1-\TV\big(P^{1,(n)}_{T,Z},P^{2,(n)}_{T,Z}\big)$, equal to
$1$ in the witness for every $n$ and decaying only as the evidence laws separate.
\textnormal{(iii)} Any rule with false-adapt and false-freeze rates $\le\alpha$ must
abstain on matched evidence with probability $\ge1-2\alpha$: abstention is mandatory,
not conservative.
\end{theorem}
\begin{proof}[Proof sketch]
(i) The witness makes $Z$ a function of $X$ alone with $P^1(X)=P^2(X)$. (ii) A committal
rule is a test between the two evidence laws; Le Cam's testing affinity plus data
processing. (iii) Matched evidence forces common action probabilities $q_a,q_f$;
validity caps each at $\alpha$. Validated: empirical $\inf_g M$ tracks $1-\TV$ across
all $n$, $100\%$ abstention on the witness.
\end{proof}

\begin{theorem}[The adapt/freeze/abstain certificate]\label{thm:cert}\label{cor:samplecomp}
\emph{Intuition.} When a benefit estimate carries a valid uncertainty radius, committing only
outside the abstain band controls false-adapt and false-freeze at level $\alpha$.
Suppose $\hat\Delta(z)$ admits a radius $\varepsilon(z)$ with
$\Pr[|\hat\Delta-\Delta|\le\varepsilon]\ge1-\alpha$. The rule
\textsc{adapt} if $\hat\Delta-\varepsilon>0$, \textsc{freeze} if
$\hat\Delta+\varepsilon<0$, else \textsc{abstain}, has marginal, unconditional false-adapt and false-freeze
rates $\le\alpha$ each (over the calibration draw under exchangeability). With $\hat\Delta$ an empirical-Bernstein mean of $n$ bounded
paired benefits (variance proxy $\sigma^2$, range $b-a$), the certificate commits once
$n\gtrsim\big(2\sigma^2/\Delta^2\big)\log(2/\alpha)+3(b-a)|\Delta|^{-1}\log(2/\alpha)$:
sample complexity scales as $\Delta^{-2}$, logarithmically in $1/\alpha$. A sequential,
anytime-valid e-process version (valid at any stopping time; $\beta$-drift-robust
variant included) is Theorem~B.1 of the manuscript.
\end{theorem}
\begin{proof}[Proof sketch]
On the good event, \textsc{adapt} implies $\Delta\ge\hat\Delta-\varepsilon>0$; the
complement has mass $\le\alpha$; commitment when $\varepsilon(n)<|\Delta|$ inverts the
Bernstein radius. Note that this marginal conformal guarantee applies to the radius and decision rather than the accuracy of the benefit model itself, and relies on calibration-deployment exchangeability at the locked-protocol granularity, degrading under task shift. The residual burden --- a calibration sample exchangeable with
deployment --- is exactly what Theorem~\ref{thm:conj1-dichotomy}(iv) makes auditable.
Validated on synthetic e-process tests (false-adapt $\le\alpha$ at nominal level).
\end{proof}

\begin{theorem}[Exact benefit-sign frontier]\label{thm:frontier}
\emph{Intuition.} On the disagreement region, the benefit sign is knowable iff an observable
margin $|M|$ exceeds a calibration-drift budget $\beta$; otherwise abstention is forced.
Fix the observables $(\mu,P_S,f_0,f_a)$ with $\mu(D)>0$, so $M$ is determined, and let
$\mathcal C_\beta=\{P_T:|\gamma|\le\beta\}$. Noting that $\gamma$ is a population quantity not observed at deploy, the statements below hold \emph{unconditionally}
for \emph{every} drift budget $\beta\ge0$ and every target with $\mu(D)>0$ --- the sign reduction
of Lemma~\ref{lem:reduction} is an algebraic identity requiring no regularity --- with $\beta$ a free
parameter of the frontier rather than an assumption (and (iii) shows every identifying class is
of this form). The location/alignment regularity invoked elsewhere is required only for the
separate graded \emph{capacity} statement (Appendix~\ref{sec:knowcap}), not for the sign frontier here.
\textnormal{(i)} $\gamma$ is a one-dimensional sufficient statistic for $\sign\Delta$
given the observables.
\textnormal{(ii)} $\sign\Delta$ is identifiable over $\mathcal C_\beta$ \emph{iff}
$|M|>\beta$, and then equals $\sign M$; the minimax committal error is $0$ if
$|M|>\beta$ and $\tfrac12$ otherwise, so the three-way rule (\textsc{adapt} if
$M>\beta$, \textsc{freeze} if $M<-\beta$, else \textsc{abstain}) is the unique maximal
sound certificate.
\textnormal{(iii)} \emph{Necessity:} any class on which $\sign\Delta$ is identifiable
lies on one side of the hyperplane $\gamma=-M$; every identifying assumption is a
one-sided constraint on $\gamma$.
\textnormal{(iv)} The label-free heuristics ATC~\cite{garg2022atc}, DoC,
GDE~\cite{jiang2022gde}, COT~\cite{lu2023cot},
AETTA~\cite{lee2024aetta}, and agreement-on-the-line~\cite{baek2022aol,kim2024taol} are
exactly the $\beta{=}0$ face: sound iff $\gamma=0$, sharing the single failure mode
$\gamma\neq0$.
\end{theorem}
\begin{proof}[Proof sketch]
All from Lemma~\ref{lem:reduction}: (ii) a $\pm\beta$ perturbation crosses $0$ iff
$|M|\le\beta$, plus the two-point argument of Theorem~\ref{thm:imp}; (iii) constancy of
$\sign(M+\gamma)$ is one-sidedness; (iv) each heuristic thresholds an observable
surrogate, i.e.\ commits to $\sign M$. Validated: frontier law exact over $5\times10^5$
draws; the unconditional family characterization (identifiable over $\mathcal C_\beta$ \emph{iff}
$|M|>\beta$) holds for every tested $\beta$; ATC error matches the predicted event to machine
precision at drift $0.04$--$0.28$ (synthetic validators for Theorems T1--T5 and AGL).
\end{proof}

\begin{corollary}[Knowable regimes]\label{thm:pos}\label{cor:irreducible}
\textnormal{(a)} Under covariate shift ($P_S(Y{\mid}X)=P_T(Y{\mid}X)$, bounded density
ratio), importance weighting identifies $\Delta$ and its sign from labeled source plus
unlabeled target ($\gamma$-budget discharged with $\beta=0$ on the importance-weighted
score); the premise itself is not label-free testable, re-entering
Theorem~\ref{thm:imp} when it fails. \textnormal{(b)} Conversely the budget is
\emph{irreducible}: no label-free certificate identifies $\sign\Delta$ without a
supplied bound on $\gamma$ --- the assumption-free certificate sought by prior
heuristics does not exist. (Label shift, bounded drift, smooth drift, and the unified
ambiguity-reach hierarchy: manuscript, App.~C.)
\end{corollary}

\begin{theorem}[One-bit dichotomy; Conjecture~\ref{conj:gen} resolved]\label{thm:conj1-dichotomy}\label{thm:H-deficit}\label{sec:conj1-closure}
\emph{Intuition.} Label-free observables fix every adaptation statistic except one global
orientation bit; no testable assumption class removes that bit in full generality.
Consider a panel $f_0,f_1,\dots$ on $D$ with correctness indicators conditionally
independent given $Y$ (CEI) and per-class symmetric accuracies (jointly,
hypothesis~\textbf{H}; advantages $b_j=2a_j-1$, agreements $c_{ij}=2A_{ij}-1$; the
rank-one algebra and anchor are classical~\cite{parisi2014,jaffe2015,platanios2014,ibrahim2019} ---
new here are the deficit, the dichotomy, and the audit).
\textnormal{(i) \emph{H is minimal:}} under CEI alone,
$c_{ij}-b_ib_j=\pi(1-\pi)\delta_i\delta_j$ exactly ($\delta_j$ the per-class accuracy
gap), so the rank-one law holds for a $K\ge3$ panel iff every $\delta_j=0$.
\textnormal{(ii) \emph{One bit:}} under H the full evidence law determines every
$|b_j|$, every product $b_ib_j$, and every benefit magnitude $|b_j-b_0|$ --- everything
except a single global flip $b\mapsto-b$, which is free at \emph{every} moment order:
the label complement preserves the entire evidence law ($\TV=0$) while negating every
benefit sign. The drift $\gamma=b_a/2-M$ is thereby point-identified up to that bit.
\textnormal{(iii) \emph{No testable bracketing (the dichotomy):}} in any output space
(binary, multiclass, regression) there is a swap involution $P\mapsto P^\star$ ---
conditional mass exchange between the two disagreeing predictions on $D$, midpoint
reflection for regression --- with $\TV(L(P),L(P^\star))=0$ for the law $L$ of
\emph{all} label-free observables (labeled source included) and
$\Delta(P^\star)=-\Delta(P)$. Hence no \emph{evidence-definable} (label-free testable)
assumption class identifies $\sign\Delta$; every identifying assumption is a
\emph{bit-selector} ($\gamma$-budgets select $\sign M$; majority-above-chance selects
$\sign(\mathrm{median}\,b)$; the agreement-line slope selects $\sign(1-2\bar w)$); and
the minimal untestable supplement is exactly one bit, which suffices under H.
\textnormal{(iv) \emph{Budgets are falsifiable, never verifiable:}} the bit-robust test
rejecting ``$|\gamma|\le\beta$'' when $\min_{\mathrm{flip}}|\hat\gamma|>\beta+r_n$ is
level-$\alpha$ with power $\ge1-\alpha$ beyond $2r_n$; verification fails exactly on
the blind set $\big\{\,\big||M|-|b_a|/2\big|\le\beta<|M|+|b_a|/2\,\big\}$, and the
$M\ge4$ rank-one residual $\tau$ is sound but \emph{not} complete (strictly interior
stealth laws with $\tau=0$ exist that bias $b$ and can flip the recovered decision).
\end{theorem}
\begin{proof}[Proof sketch]
(i) Expand $c_{ij}$ over $Y$ and cancel cross terms. (ii) Triple products
$c_{ik}c_{il}/c_{kl}=b_i^2$; the complement $Y'=1-Y$ fixes predictions (maps of $X$)
hence the whole evidence law, while $b\mapsto-b$ and H is preserved. (iii) The swap
preserves the $X$-marginal and source, so $L$ is fixed at every order; on $D$ it
exchanges $p_a\leftrightarrow p_0$ (or negates $f_0{+}f_a{-}2Y$), so $\Delta$ negates;
identifiability on a swap-closed class is impossible, and selecting one member per swap
orbit is by definition one bit. Algebraically: observables generate the \emph{even}
correctness subalgebra ($s_is_j=u_iu_j$), the decision is \emph{odd} ($s_j=u_jv$), and
the swap is $v\mapsto-v$. (iv) Empirical-Bernstein for $\hat M$, a product-ratio
perturbation bound for $\hat b_a$, and the flip-min for bit-robust soundness; the
stealth construction is a strictly interior LP on the $2^4$ correctness patterns
matching all pairwise moments to a rank-one target with a shifted first moment.
Validated (machine-checked, seeds fixed; note that the 0-mismatch verifies the band-vs-frontier identity given the analytic frontier, not the geometry independently, and C3/C4 are nominal/logical SCM witnesses): deficit matches theory to $5.7\times10^{-4}$;
flip $\TV=0$ with $\max_j|b'_j+b_j|=1.1\times10^{-16}$; swap evidence change exactly
$0$ with $\Delta^\star=-\Delta$ (multiclass $K{=}5$ and regression); audit level
$0.000\le\alpha{=}0.05$, power $1.000$; stealth $\tau=2\times10^{-17}$ with recovery
bias $0.229$ (Figs.~\ref{fig:onebit}--\ref{fig:conj1closure}).
Full proofs appear in the supplementary theory appendix.
\end{proof}

\begin{conjecture}[Label-free bracketing --- \textbf{resolved}]\label{conj:gen}
The unconditional label-free bracketing of the ordinal comparison on $D$ (posed as the
open piece in earlier versions) \emph{does not exist}: by
Theorem~\ref{thm:conj1-dichotomy}(iii) no evidence-definable assumption identifies
$\sign\Delta$, and the minimal untestable supplement is one bit. The conditional
resolutions --- calibration transfer, the $\gamma$-meter under checkable agreement
structure, H plus an anchor (all in the manuscript) --- are therefore of the minimal
possible form: \emph{falsifiable structure plus one declared bit}. On the
\emph{margin-monotone relative-calibration} class $\mathcal C_{\mathrm{mono}}$
(Definition~\ref{def:cmono}) one declared bit certifies $\sign\Delta$ \emph{unconditionally}
(Theorem~\ref{thm:cmono-weakest}(i)--(ii)); and $\mathcal C_{\mathrm{mono}}$ is \emph{a} weakest
such class under a general-position assumption (Theorem~\ref{thm:cmono-weakest}(iii)). The
\emph{unconditional} weakest classes (general position removed) are characterized as an explicit
finite family of dominance polytopes (Theorem~\ref{thm:uncond-weakest}); there is no unique weakest
class, consistent with the bracketing non-existence above.
\end{conjecture}

\begin{theorem}[The evidence-channel rate: labels buy constants and the bit, not the rate]\label{thm:ev-rate}\label{thm:agl}
\emph{Intuition.} Unlabeled agreement statistics and labeled paired benefits share the same
$1/\sqrt{m}$ rate; labels change constants and supply the missing bit, not the exponent.
Within audited H with margins ($\min_{i\ne j}|c_{ij}|\ge c_{\min}$,
$\min_j|b_j|\ge b_{\min}$) and a declared bit, $m$ unlabeled samples on $D$ give
$|\hat b_j-b_j|\le C\,\sqrt{\log(K/\delta)/m}$ with
$C=\Theta\big(1/(b_{\min}c_{\min}^2)\big)$, so $\sign(b_j-b_0)$ is certified at
$m\asymp C^2(b_j-b_0)^{-2}\log(K/\delta)$; a Le Cam two-point bound inside H shows
$m^{-1/2}$ is minimax for the evidence channel. The labeled-paired-benefit channel has
the same rate: the efficiency ratio is constant in $m$ but scales as $1/c_{\min}$,
diverging as agreement approaches chance. Relatedly, accuracy is an exact affine
function of agreement-with-reference iff the disagreement win rate is constant (slope
$1-2\bar w$), the anchored form of agreement-on-the-line; its slope \emph{sign} is the
same bit as in Theorem~\ref{thm:conj1-dichotomy}(iii) (manuscript, \S AGL).
\end{theorem}
\begin{proof}[Proof sketch]
Hoeffding on agreements, the product-ratio perturbation lemma, and a $\chi^2$/KL bound
between $K$-bit pattern laws differing by $\varepsilon$ in one coordinate
(KL$\,\le C'\varepsilon^2$), tensorized; Bretagnolle--Huber finishes. Validated:
recovery slope $-0.51$; minimax slope $-0.525$ tracking $c_0/\sqrt m$;
KL$/\varepsilon^2\in[0.164,0.171]$; efficiency $2.15\to4.49\to\infty$ as
$c_{\min}\,0.21\to0.067\to{\le}0.022$ (Fig.~\ref{fig:evrate};
AGL: $R^2=1.000$ under constant $w$, slope $1-2w$ exact on synthetic checks).
\end{proof}

\smallskip
\noindent\emph{The weakest one-bit class.} Theorem~\ref{thm:conj1-dichotomy} fixes the supplement at one bit; the \emph{weakest falsifiable class} on which that bit suffices is settled conditionally ---a margin-monotone relative-calibration class on which one declared bit certifies $\sign\Delta$ unconditionally, and \emph{a} weakest such class under a general-position assumption (Appendix~\ref{app:weakest}, Thm~\ref{thm:cmono-weakest}). The \emph{unconditional} weakest classes form an explicit finite family of dominance polytopes (Thm~\ref{thm:uncond-weakest}).
