\begin{lemma}[Disagreement-region reduction]\label{lem:reduction}
Under Assumption~\ref{ass:deploy}, with $\bar a:=\Pr_T(f_a{=}Y\mid D)$,
\begin{align*}
\Delta &= 2\,\mu_T(D)\bigl(\bar a-\tfrac12\bigr),\\
\sign\Delta &= \sign(\bar a-\tfrac12)=\sign(M+\gamma).
\end{align*}
The identity $\sign\Delta=\sign(M+\gamma)$ holds for \emph{every} target law; it does not
assume $|\gamma|\le\beta$.
\end{lemma}
\begin{proof}
On $D$ exactly one of $f_0,f_a$ is correct under $0/1$ loss, so the benefit equals
$2\mu_T(D)(\bar a-\tfrac12)$. By the definitions of $M$ and $\gamma$,
\begin{align*}
M+\gamma
&=\Bigl(\E_{\mu_T\mid D}[s]-\tfrac12\Bigr)
  +\E_{\mu_T\mid D}[\eta_a-s]\\
&=\E_{\mu_T\mid D}[\eta_a]-\tfrac12
 =\bar a-\tfrac12,
\end{align*}
so $\sign\Delta=\sign(\bar a-\tfrac12)=\sign(M+\gamma)$.
\end{proof}

\begin{theorem}[Interior matched-evidence impossibility]\label{lem:nonid}
Fix $\beta>0$. For any margin with $|M|<\beta$ there exist
$P_T^1,P_T^2\in\mathcal{C}_\beta$ such
that $\Law(\widetilde Z\mid P_T^1)=\Law(\widetilde Z\mid P_T^2)$ for
$\widetilde Z=(Z,M)$, $M(P_T^1)=M(P_T^2)=M$, and
$\sign\Delta_1=-\sign\Delta_2\neq0$. (For $\beta=0$ the class
$\mathcal{C}_0$ forces $\gamma=0$, so $\sign\Delta=\sign M$
is determined and no such pair exists: matched-evidence ambiguity is a strictly-positive-drift
phenomenon.)
\end{theorem}
\begin{proof}
Choose $0<\delta<\min\{\beta-|M|,\tfrac12\}$ and construct the two label laws explicitly. For
$\theta\in\{+1,-1\}$ define $P_T^\theta$ on $\mathcal X\times\{0,1\}$ by
$P_T^\theta(dx,y)=\mu_T(dx)\,K^\theta(x,y)$ with label kernel
\[
K^\theta(x,y)=
\begin{cases}
\tfrac12+\theta\delta, & x\in D,\ y=f_a(x),\\[2pt]
\tfrac12-\theta\delta, & x\in D,\ y=f_0(x),\\[2pt]
K^{\circ}(x,y), & x\notin D,
\end{cases}
\]
where $K^{\circ}$ is any fixed conditional label law used by \emph{both} worlds (e.g.\ the source
conditional). On $D$ the binary labels satisfy $\{f_0(x),f_a(x)\}=\{0,1\}$, so $K^\theta(x,\cdot)$
is a valid Bernoulli law ($\delta<\tfrac12$ keeps both masses in $(0,1)$); off $D$ the two
predictors agree, so those labels contribute identically to both risks and cancel in $\Delta$.
\emph{Class membership:} $\eta_a^\theta(x)=\tfrac12+\theta\delta$ on $D$, hence
$\bar a_\theta=\tfrac12+\theta\delta$ and
$\gamma_\theta=\bar a_\theta-(M+\tfrac12)=\theta\delta-M$, with
$|\gamma_\theta|\le\delta+|M|<\beta$; both worlds lie in $\mathcal{C}_\beta$.
\emph{Matched evidence:} the input marginal $\mu_T$ and $(f_0,f_a,s)$ are unchanged, and
$Z=\phi(X_{1:m},f_0,f_a)$ uses no labels, so
$\Law(\widetilde Z\mid P_T^{+})=\Law(\widetilde Z\mid P_T^{-})$ for $\widetilde Z=(Z,M)$.
\emph{Opposite signs:} Lemma~\ref{lem:reduction} gives
$\Delta_\theta=2\mu_T(D)\,\theta\delta\neq0$ with $\sign\Delta_\theta=\theta$.
This completes the construction.
\end{proof}
\noindent The two evidence laws here coincide \emph{exactly}, not merely approximately: the
construction changes only the label kernel on $D$, and $Z$ is a function of unlabeled inputs and
model outputs. We therefore do not present this as a Le Cam two-point argument. A Le Cam argument
trades off a positive total-variation distance against the achievable error; here the distance is
zero, which is the degenerate and strongest case, and the corresponding statement is exact
non-identifiability rather than a rate.

\begin{corollary}[Matched-evidence abstention lower bound]\label{cor:matched-abstain}
Fix $\alpha\in(0,\tfrac12)$. On the shared evidence law constructed in
Theorem~\ref{lem:nonid}, any possibly randomized label-free rule $g$ satisfying
$\Pr[g=\textsc{adapt},\Delta\le0]\le\alpha$ in every admissible world and
$\Pr[g=\textsc{freeze},\Delta\ge0]\le\alpha$ in every admissible world must satisfy
$\Pr[g=\textsc{abstain}]\ge1-2\alpha$.
\end{corollary}
\begin{proof}
The rule has the same action probabilities in the two evidence-identical worlds. Its adapt
probability is at most $\alpha$ in the negative world and its freeze probability is at most
$\alpha$ in the positive world. The three action probabilities sum to one.
\end{proof}

\begin{proposition}[Boundary case and closed-band abstention]\label{prop:closed-band}
Fix $\alpha\in(0,\tfrac12)$. Under Definition~\ref{def:strict-sound}, abstention is the pointwise
maximal sound three-way action throughout the closed band $|M|\le\beta$, and any label-free rule
controlling both marginal errors at level $\alpha$ abstains there with probability at least
$1-2\alpha$. Opposite nonzero signs are asserted only in the interior $|M|<\beta$.
\end{proposition}
\begin{proof}
For $|M|<\beta$ this is Theorem~\ref{lem:nonid} together with
Corollary~\ref{cor:matched-abstain}. For $|M|=\beta>0$, the declared class contains an
augmented-evidence-identical zero-benefit world (take $\gamma=-M$, admissible since
$|\gamma|=\beta$) and a strict-benefit world of sign $M$ (take $\gamma=0$). In the zero-benefit world
$\Delta=0$, so \emph{both} directional error events $\{\textsc{adapt},\Delta\le0\}$ and
$\{\textsc{freeze},\Delta\ge0\}$ occur whenever the rule commits; since the rule's action
probabilities are identical across the two evidence-identical worlds, dual control at level $\alpha$
gives $\Pr[\textsc{adapt}]\le\alpha$ and $\Pr[\textsc{freeze}]\le\alpha$, hence
$\Pr[\textsc{abstain}]\ge1-2\alpha$, and neither strict action is uniformly directionally sound.
For $M=\beta=0$ the class $\mathcal C_0$ forces $\gamma=0$, so $\Delta\equiv0$ and the same
conclusion holds directly.
\end{proof}
\noindent This directional convention is stricter
than task regret: at $\Delta=0$, adapt and freeze are risk-equivalent but neither certifies a strict
sign.

\begin{theorem}[Exact strict-commitment frontier]\label{thm:headline}\label{thm:frontier}
Under Assumption~\ref{ass:deploy}, fix the observables $(\mu_T,P_S,f_0,f_a,s)$ so $M$ is
determined, and let $\beta\ge0$. The necessity and maximality clauses use the full class
$\mathcal C_\beta$ and therefore its richness: it contains the evidence-identical label kernels
constructed in Theorem~\ref{lem:nonid} and the zero-benefit boundary law. For a narrower declared
subclass not closed under these constructions, only the sufficiency clause follows without an
additional richness assumption.
\begin{enumerate}
\item[\textnormal{(i)}] \textbf{Reduction.} For every $P_T$, $\sign\Delta=\sign(M+\gamma)$
(Lemma~\ref{lem:reduction}).
\item[\textnormal{(ii)}] \textbf{Sufficiency.} If $|M|>\beta$ and $P_T\in\mathcal{C}_\beta$, then
$\sign\Delta=\sign M$.
\item[\textnormal{(iii)}] \textbf{Necessity.} If $|M|<\beta$ (which requires $\beta>0$), there exist
$P_T^1,P_T^2\in\mathcal{C}_\beta$ with the same augmented-evidence law and opposite \emph{nonzero} benefit
signs; at $|M|=\beta>0$ the admissible drift $\gamma=-\sign(M)\beta$ yields $\Delta=0$, and in the
degenerate case $M=\beta=0$ the class $\mathcal{C}_0$ forces $\Delta\equiv0$. Hence no \emph{strict}
benefit direction is uniformly supported by $\Law(\widetilde Z)$ over $\mathcal{C}_\beta$
whenever $|M|\le\beta$.
\item[\textnormal{(iv)}] \textbf{Strict-commitment iff.} At the fixed augmented-evidence law, a
strict action in $\{\textsc{adapt},\textsc{freeze}\}$ is uniformly directionally sound in the
sense of Definition~\ref{def:strict-sound} if and only if
$|M|>\beta$; on the closed band $|M|\le\beta$ (including $M=\beta=0$, where $\Delta\equiv0$) abstention
is the pointwise maximal sound action.
\end{enumerate}
The population rule $M>\beta\Rightarrow\textsc{adapt}$, $M<-\beta\Rightarrow\textsc{freeze}$,
$|M|\le\beta\Rightarrow\textsc{abstain}$ is the pointwise maximal sound certificate on
$\mathcal{C}_\beta$.
\end{theorem}
\begin{proof}
Part~(i) is Lemma~\ref{lem:reduction}. For~(ii), if $M>\beta$ and $|\gamma|\le\beta$, then
$M+\gamma>0$, so $\sign\Delta=\sign(M+\gamma)=+1$; the negative case is symmetric. For~(iii),
Theorem~\ref{lem:nonid} supplies two laws in the same augmented-evidence fibre with opposite
nonzero signs whenever $|M|<\beta$. At $|M|=\beta$ the boundary law with
$\gamma=-\sign(M)\beta$ gives $\Delta=0$, still precluding a determined nonzero sign. Part~(iv)
combines (ii), (iii), and Proposition~\ref{prop:closed-band}. Pointwise maximality follows because
(ii) certifies the unique strict direction outside the band, while Theorem~\ref{lem:nonid} and
Proposition~\ref{prop:closed-band} rule out both strict directions inside and on the boundary.
\end{proof}

\begin{theorem}[Finite-sample adapt/freeze/abstain certificate]\label{thm:certificate}\label{thm:cert}
Suppose $\Delta$, $\widehat\Delta$, and the nonnegative random radius $\varepsilon$ are measurable
on a common probability space containing all calibration, benefit-model fitting, and
deployment-unit randomness, and that the benefit interval satisfies marginal coverage
\[
\Pr\!\bigl[\,|\widehat\Delta-\Delta|\le\varepsilon\,\bigr]\ \ge\ 1-\alpha .
\]
Define
\[
g(Z)=\begin{cases}
\textsc{adapt} & \widehat\Delta-\varepsilon>0,\\
\textsc{freeze} & \widehat\Delta+\varepsilon<0,\\
\textsc{abstain} & \text{otherwise}.
\end{cases}
\]
Then, marginally over that common probability space,
\begin{align*}
\Pr\bigl[g(Z)=\textsc{adapt},\Delta\le0\bigr]&\le\alpha,\\
\Pr\bigl[g(Z)=\textsc{freeze},\Delta\ge0\bigr]&\le\alpha.
\end{align*}
\end{theorem}
\begin{proof}
On the event $\{|\widehat\Delta-\Delta|\le\varepsilon\}$, the condition
$\widehat\Delta-\varepsilon>0$ implies $\Delta>0$, so a false adapt requires the complement of
this event, which has probability at most $\alpha$. The freeze case is symmetric.
\end{proof}
Exchangeability, or an explicit shift correction, is one route to this coverage assumption. Risk
alignment is not required for the conditional-on-coverage implication.

\begin{remark}[Marginal vs.\ conditional error]\label{rem:fa-marginal}
Theorem~\ref{thm:certificate} bounds the \emph{marginal, unconditional} false-adapt and
false-freeze probabilities ($\mathrm{FA}_{\mathrm u}$). It does \emph{not} bound the conditional
rate $\mathrm{FA}_{\mathrm c}=\Pr[\Delta\le0\mid\textsc{adapt}]$, which we report empirically.
Risk alignment (Definition~\ref{def:risk-align}) is required for the \emph{frontier}
identifiability claim; coverage alone certifies the \emph{interval} rule above.
\end{remark}

\begin{remark}[Fallback when assumptions are unsupported]\label{cor:abstain-valid}
If interval coverage---or the calibration assumptions supporting it (exchangeability, or an explicit
shift correction)---cannot be justified on the deployment batch, Theorem~\ref{thm:certificate}
supplies no level-$\alpha$ protection for either strict action. A deployment must then use a
separately declared operational fallback, such as abstention or serving the frozen model; this is a
policy recommendation, not a conclusion that either fallback is optimal. If instead \emph{risk alignment}
(Definition~\ref{def:risk-align}) cannot be justified, the population-frontier interpretation and the
transferability of the benefit estimator are unsupported, even though the interval implication of
Theorem~\ref{thm:certificate} remains valid \emph{conditional on coverage} (cf.\ Theorem~\ref{lem:nonid}).
\end{remark}
