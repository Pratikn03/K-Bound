\section{Auditable drift budgets: what can replace a declared $\beta$}\label{app:auditable}

The main text treats the drift budget $\beta$ as \emph{declared}. This appendix settles when
$\beta$ can instead be \emph{computed}: a vacuity theorem (label-free audits certify nothing), three
purchasable audits (labels; smoothness$\times$transport; rotation-uniform under unknown-direction
index drift), a witness-class sufficient condition for useful root-$n$ audits
(Theorem~\ref{thm:aud-H}), the composition into the decision guarantee, a fully empirical rule, and a
domain-level verifiability protocol with an exact feasibility floor. Notation as in
Assumption~\ref{ass:deploy}: $u=\eta_a-s\in[-1,1]$ on $D$, $\gamma(P)=\E_{\mu_P}[u\mid D]$;
representations $\psi(x)$ with $\|\psi\|\le B$; all statements conditional on $D$.

\paragraph{Status and credit.} Proofs are complete below; the concentration ingredient of
Theorem~\ref{thm:aud-F} is \emph{known} in sharper form (Bartl--Mendelson; Boedihardjo; Nietert
et al.)~\cite{bartl2022structure,bartl2023uniformdkw,boedihardjo2024sharp,nietert2022sliced} and
is used as a lemma; the hierarchical-conformal backbone of Theorem~\ref{thm:aud-G} follows
Dunn--Wasserman--Ramdas and Lee--Barber--Willett~\cite{dunn2022hierarchical,lee2023hierarchical}.
The contribution is the composition into the adapt/freeze/abstain audit, the witness-class
sufficient condition (Theorem~\ref{thm:aud-H}), and the floor analysis; synthetic validation
(coverage, non-vacuity, decision safety, Aud-F net panel) is in the repository artifact
\texttt{audit\_validation/results\_audited\_drift.json}.

\begin{theorem}[Aud-A: label-free budget audits are vacuous]\label{thm:aud-A}
Fix the observables $(\mu_T,f_0,f_a,s)$, hence $M$, and let $\mathcal P(\mu_T)$ be all target
laws with these observables. For every $\bar a\in[0,1]$ there is a law in $\mathcal P(\mu_T)$
with $\gamma=\bar a-\tfrac12-M$ and the same joint law of $(Z,M)$; the achievable drift set is
$[-(M+\tfrac12),\,\tfrac12-M]$ at matched evidence. Consequently any audit
$\widehat\beta=g(Z)$ with $\Pr(|\gamma|\le\widehat\beta)\ge1-\delta$ uniformly over
$\mathcal P(\mu_T)$ satisfies, with probability $\ge1-\delta$,
$\widehat\beta\ge\tfrac12+|M|>|M|$: the audited frontier abstains with probability $\ge1-\delta$
in every world.
\end{theorem}
\begin{proof}
Extend the kernel construction in the proof of Theorem~\ref{lem:nonid}: for $\bar a\in[0,1]$ set
$K^{\bar a}(x,f_a(x))=\bar a$, $K^{\bar a}(x,f_0(x))=1-\bar a$ on $D$ (valid Bernoulli since
binary labels give $\{f_0(x),f_a(x)\}=\{0,1\}$ on $D$), keeping the off-$D$ kernel and all
observables fixed. Then $\Law(Z,M)$ is identical across $\bar a$ while
$\gamma=\bar a-\tfrac12-M$ sweeps the stated interval. Validity in the extreme world
($\bar a\in\{0,1\}$ as $\sign M$ dictates, where $|\gamma|=\tfrac12+|M|$) forces the
$(1-\delta)$-quantile of $g(Z)$ to be at least $\tfrac12+|M|$; since $g(Z)$ has the same
distribution in every world of $\mathcal P(\mu_T)$, the bound transfers, and on that event
neither strict branch of $|M|>\widehat\beta$ can fire.
\end{proof}

\begin{theorem}[Aud-B: label-budget audit]\label{thm:aud-B}
Let $X_1,\dots,X_n\sim\mu_T(\cdot\mid D)$ with labels revealed, and
$\widehat u_i=\mathbf 1\{f_a(X_i)=Y_i\}-s(X_i)\in[-1,1]$,
$\widehat\gamma=\tfrac1n\sum\widehat u_i$, $t(n,\delta)=\sqrt{2\ln(2/\delta)/n}$. Then
$\widehat\beta_{\mathrm{lab}}=|\widehat\gamma|+t(n,\delta)$ satisfies
$\Pr(|\gamma|\le\widehat\beta_{\mathrm{lab}})\ge1-\delta$, and the audited frontier certifies a
strict action whenever $|M|>|\gamma|+2t(n,\delta)$; in particular
$n\ge 8\ln(2/\delta)/(|M|-|\gamma|)^2$ audit labels suffice when $|M|>|\gamma|$.
\end{theorem}
\begin{proof}
$\E[\widehat u_i]=\gamma$; Hoeffding on range-2 variables gives
$\Pr(|\widehat\gamma-\gamma|>t)\le2e^{-nt^2/2}=\delta$; the triangle inequality gives both
claims. (This quantifies, in the form the frontier consumes, the audit rate discussed in
App.~\ref{app:weakest}.)
\end{proof}

\begin{theorem}[Aud-C: transport-audited budget, finite-sample]\label{thm:aud-C}
Assume \textup{(Lip)}: $u$ is $L$-Lipschitz on $D$ w.r.t.\ a bounded 1-d representation
($|\psi|\le B$). With $n_{\mathrm{lab}}$ labeled calibration points, $m_{\mathrm{cal}},
m_{\mathrm{dep}}$ unlabeled points and empirical measures $\widehat\mu$, let
$\varepsilon_{\mathrm{DKW}}(m,\delta)=\sqrt{\ln(2/\delta)/(2m)}$ and define
\begin{align*}
\widehat\beta_{W}
&=|\widehat\gamma_{\mathrm{cal}}|+t(n_{\mathrm{lab}},\delta/2)\\
&\quad+L\widehat W_1(\widehat\mu_T,\widehat\mu_{\mathrm{cal}})\\
&\quad+2BL\varepsilon_{\mathrm{DKW}}(m_{\mathrm{cal}},\delta/4)\\
&\quad+2BL\varepsilon_{\mathrm{DKW}}(m_{\mathrm{dep}},\delta/4).
\end{align*}
we have $\Pr(|\gamma(P_T)|\le\widehat\beta_{W})\ge1-\delta$.
\end{theorem}
\begin{proof}
Population step: $u/L$ is 1-Lipschitz, so Kantorovich--Rubinstein gives
$|\gamma(P_T)-\gamma(P_{\mathrm{cal}})|\le L\,W_1(\mu_T,\mu_{\mathrm{cal}})$. Finite-sample:
on a 1-d bounded representation, $W_1(\widehat\mu,\mu)=\int|\widehat F-F|\le
2B\sup_t|\widehat F-F|$, bounded by DKW at level $\delta/4$ per measure; Hoeffding at
$\delta/2$ for $\widehat\gamma_{\mathrm{cal}}$; union bound and the $W_1$ triangle inequality
assemble $\widehat\beta_W$. \textup{(Lip)} is declared, not verified---Theorem~\ref{thm:aud-A}
guarantees some such purchase is unavoidable; multi-domain Lipschitz-ratio checks falsify but
cannot verify it.
\end{proof}

\begin{theorem}[Aud-D/E: composition and the fully empirical rule]\label{thm:aud-DE}
Let $\widehat\beta$ be any audit with $\Pr(|\gamma|\le\widehat\beta)\ge1-\delta$.
\textup{(D)}~The audited population rule (adapt iff $M>\widehat\beta$; freeze iff
$M<-\widehat\beta$; else abstain) has $\Pr(\text{false strict commitment})\le\delta$.
\textup{(E)}~With $\widehat M=\tfrac1m\sum s(X_i)-\tfrac12$ from $m$ unlabeled deployment
points and $t_M(m,\delta')=\sqrt{\ln(2/\delta')/(2m)}$, the \emph{empirical} audited rule
(adapt iff $\widehat M-t_M>\widehat\beta$; freeze iff $\widehat M+t_M<-\widehat\beta$) has
$\Pr(\text{false strict commitment})\le\delta+\delta'$, and every quantity in the rule is
computable. Gated by the finite-sample certificate of Theorem~\ref{thm:certificate}, each
false-commit event is bounded by $\alpha+\delta(+\delta')$.
\end{theorem}
\begin{proof}
(D): on $\{|\gamma|\le\widehat\beta\}$, $M>\widehat\beta\ge|\gamma|$ implies $M+\gamma>0$, so
$\sign\Delta=+1$ by Lemma~\ref{lem:reduction}; symmetric for freeze; a false commitment
requires the audit event to fail. (E): Hoeffding for the $[0,1]$-valued $s(X_i)$ gives
$\Pr(|\widehat M-M|>t_M)\le\delta'$; on the intersection of the two events the adapt branch
gives $M\ge\widehat M-t_M>\widehat\beta\ge|\gamma|$; union bound. Validation: on the synthetic
sweep the empirical rule makes $0/12$ false strict commits (population-$M$ rule $0/13$;
$\beta{=}0$ plug-in $3/20$), with certification opening one grid step later than the
population rule---the price of $t_M$
(\texttt{audit\_validation/results\_audited\_drift.json}).
\end{proof}

\paragraph{Rotation-closed classes and useful audits.}
The maximality questions below need two notions that are used throughout the rest of the
appendix.
\begin{definition}[Rotation-closed drift class]\label{def:rotation-closed}
Fix a bounded feature map $\|\psi(x)\|\le B$ on $D$. A class $\mathcal U$ of drift functions
$u:D\to[-1,1]$ is \emph{rotation-closed} if $u\in\mathcal U$ implies
$u\circ R_\psi\in\mathcal U$ for every orthogonal transformation $R$ acting on the feature
space (equivalently: if a single-index ridge $u(x)=g(\theta^\top\psi(x))$ is admitted for some
$\theta\in S^{d-1}$, then the same ridge is admitted for every $\theta\in S^{d-1}$).
\end{definition}
\begin{definition}[Useful audit at rate $r_n$]\label{def:useful-audit}
An audit $\widehat\beta$ is \emph{valid} at level $\delta$ over a class $\mathcal P$ if
$\Pr(|\gamma(P)|\le\widehat\beta)\ge1-\delta$ for every $P\in\mathcal P$. It is
\emph{useful at rate $r_n$} if it is valid and, whenever the true drift vanishes
($\gamma\equiv0$ and $\mu_T=\mu_{\mathrm{cal}}$), one has $\widehat\beta=O_P(r_n)$.
Usefulness rules out the vacuous label-free bound $\widehat\beta\ge\tfrac12+|M|$ of
Theorem~\ref{thm:aud-A}.
\end{definition}

\begin{theorem}[Aud-F: rotation-uniform audit under unknown-direction index drift]\label{thm:aud-F}
Assume $\|\psi(x)\|\le B$ on $D$ and \textup{(Idx)}: $u(x)=g(\theta_*^{\!\top}\psi(x))$ with
$g$ $L$-Lipschitz and $\theta_*\in S^{d-1}$ \emph{unknown}. Let
$\mathrm{MS}_{\mathrm{net}}$ be the maximum of the empirical sliced distances
$W_1(\theta\#\widehat\mu_{\mathrm{cal}},\theta\#\widehat\mu_T)$ over an $\epsilon$-net of
$S^{d-1}$, and
where
$\varepsilon_{\mathrm{VC}}(n,\delta)=
\sqrt{2((d{+}1)\ln(2n)+\ln(8/\delta))/n}$, define
\begin{align*}
\widehat\beta_{F}
&=|\widehat\gamma_{\mathrm{cal}}|+t(n_{\mathrm{lab}},\delta/4)\\
&\quad+L(\mathrm{MS}_{\mathrm{net}}+2B\epsilon)\\
&\quad+2BL\varepsilon_{\mathrm{VC}}(n,\delta/4)
 +2BL\varepsilon_{\mathrm{VC}}(m,\delta/4).
\end{align*}
Then $\Pr(|\gamma(P_T)|\le\widehat\beta_{F})\ge1-\delta$. The same statement holds for
$k$-index drift $u=\sum_{j\le k}g_j(\theta_j^{\!\top}\psi)$ with $\sum_j\mathrm{Lip}(g_j)\le L$.
\end{theorem}
\begin{proof}
(i)~\emph{Index transfer:} $g/L$ is 1-Lipschitz on the projected line, so 1-d
Kantorovich--Rubinstein gives
$|\gamma(P_T)-\gamma(P_{\mathrm{cal}})|\le L\,W_1(\theta_*\#\mu_{\mathrm{cal}},\theta_*\#\mu_T)
\le L\,\sup_{\theta}W_1(\theta\#\mu_{\mathrm{cal}},\theta\#\mu_T)=L\cdot\mathrm{MS}$; the
supremum covers the unknown $\theta_*$.
(ii)~\emph{Uniform deviation:} for every $\theta$ simultaneously, deviations of the empirical
sliced distances are controlled by
$W_1(\widehat\nu,\nu)=\int_{[-B,B]}|\widehat F-F|\le 2B\sup_t|\widehat F-F|$ (the 1-d identity,
valid with atoms), and $\sup_{\theta,t}|\widehat F_\theta(t)-F_\theta(t)|$ is the uniform
deviation over the halfspace class $\{x:\psi(x)\cdot\theta\le t\}$, of VC dimension $\le d+1$;
the VC inequality at level $\delta/4$ per measure gives the two
$\varepsilon_{\mathrm{VC}}$ terms via $|\sup_\theta a-\sup_\theta b|\le\sup_\theta|a-b|$ and the
per-$\theta$ $W_1$ triangle inequality.
(iii)~\emph{Net:} $|W_1(\theta\#\cdot)-W_1(\theta'\#\cdot)|\le 2B\|\theta-\theta'\|$ (identity
coupling and Cauchy--Schwarz, both measures), so
$\mathrm{MS}\le\mathrm{MS}_{\mathrm{net}}+2B\epsilon$; the net-inflated value enters
$\widehat\beta_F$.
(iv)~\emph{Budget:} three random events (Hoeffding; VC for calibration; VC for deployment) at
$\delta/4$ each, total failure $\le3\delta/4<\delta$. For $k$-index, bound each term by its own
slice and use $\sum_j\mathrm{Lip}(g_j)\le L$.
Sharper constants (and removal of the $\ln(2n)$ factor) follow by replacing the VC step with
the max-sliced concentration of~\cite{bartl2022structure,boedihardjo2024sharp}; the
$O((3/\epsilon)^d)$ net is computational only---the statistical bound is net-free.
\end{proof}
\begin{remark}[Aud-F audits adaptive max-sliced $W_1$]\label{rem:aud-F-msw}
The population quantity controlled by Theorem~\ref{thm:aud-F} is the
\emph{max-sliced} (projection-robust) distance
$\Delta_1(\mu_{\mathrm{cal}},\mu_T)=\sup_{\theta\in S^{d-1}}W_1(\theta\#\mu_{\mathrm{cal}},\theta\#\mu_T)$,
not a predeclared finite set of directions. The unknown ridge $\theta_*$ is therefore
covered adaptively; the $\epsilon$-net enters only to produce a
\emph{computable} upper certificate $\mathrm{MS}_{\mathrm{net}}+2B\epsilon$ on
$\Delta_1(\widehat\mu_{\mathrm{cal}},\widehat\mu_T)$. Sharp $n^{-1/2}$ MSW$_1$ bounds under
bounded support are available from~\cite{boedihardjo2024sharp,bartl2022structure}; Aud-F's
contribution is the composition into a useful drift-budget audit for the adapt/freeze/abstain
frontier. Numerical coverage of the net certificate on unknown-direction worlds is recorded in
\texttt{audit\_validation/results\_audited\_drift.json} (panel \texttt{F\_index\_msw}).
\end{remark}

\begin{theorem}[Aud-H: useful audits for low-complexity witness classes]\label{thm:aud-H}
Let $\mathcal F$ be a symmetric class of measurable functions $D\to[-L,L]$ that contains the
unknown drift $u$ and has Rademacher complexity satisfying
$\mathfrak R_n(\mathcal F)\le C\sqrt{(v+\ln(2/\delta))/n}$ for constants $C,v$
(alternatively: VC-subgraph dimension at most $v$, which yields the same root-$n$ scale).
Define the integral probability metric
$\Delta_{\mathcal F}(P,Q)=\sup_{f\in\mathcal F}|Pf-Qf|$. With $n_{\mathrm{lab}}$ labeled
calibration points, $n$ unlabeled calibration points, and $m$ unlabeled deployment points,
set
\begin{align*}
\widehat\beta_{\mathcal F}
&=|\widehat\gamma_{\mathrm{cal}}|+t(n_{\mathrm{lab}},\delta/3)
 +\Delta_{\mathcal F}(\widehat\mu_{\mathrm{cal}},\widehat\mu_T)\\
&\quad+2\mathfrak R_n(\mathcal F)+2\mathfrak R_m(\mathcal F)\\
&\quad+L\sqrt{\tfrac{\ln(6/\delta)}{n}}
 +L\sqrt{\tfrac{\ln(6/\delta)}{m}}.
\end{align*}
Then $\Pr(|\gamma(P_T)|\le\widehat\beta_{\mathcal F})\ge1-\delta$. In particular the audit is
useful at rate $O(\sqrt{(v+\ln(1/\delta))/n})$ whenever $\mu_T=\mu_{\mathrm{cal}}$ and
$\gamma=0$. Specializations:
\textup{(i)}~projected $1$-Lipschitz / halfspace-CDF witnesses recover the adaptive MSW$_1$
budget of Theorem~\ref{thm:aud-F} with $v\asymp d$;
\textup{(ii)}~$s$-sparse ridge witnesses replace $d$ by $s\log(ed/s)$;
\textup{(iii)}~the full $1$-Lipschitz class on a $d$-dimensional ball has metric entropy
forcing the $n^{-1/d}$ empirical-$W_1$ rate~\cite{fournier2015rate,weed2019sharp} and is
therefore \emph{not} useful at $O(\sqrt{d/n})$.
\end{theorem}
\begin{proof}
(i)~\emph{Population transfer.} Since $u\in\mathcal F$ and $\mathcal F=-\mathcal F$,
\[
|\gamma(P_T)-\gamma(P_{\mathrm{cal}})|
=\bigl|\E_{\mu_T}[u]-\E_{\mu_{\mathrm{cal}}}[u]\bigr|
\le\Delta_{\mathcal F}(\mu_T,\mu_{\mathrm{cal}}).
\]
(ii)~\emph{Empirical IPM.} The triangle inequality gives
$\Delta_{\mathcal F}(\mu_T,\mu_{\mathrm{cal}})
\le\Delta_{\mathcal F}(\widehat\mu_T,\widehat\mu_{\mathrm{cal}})
+\Delta_{\mathcal F}(\widehat\mu_T,\mu_T)
+\Delta_{\mathcal F}(\widehat\mu_{\mathrm{cal}},\mu_{\mathrm{cal}})$.
Standard symmetrization yields
$\E\Delta_{\mathcal F}(\widehat\mu,\mu)\le 2\mathfrak R_n(\mathcal F)$; a bounded-difference /
McDiarmid upgrade (range $2L/n$ per sample) adds the two
$L\sqrt{\ln(6/\delta)/(2n)}$ concentration terms at total failure $\delta/3$.
(iii)~\emph{Labeled calibration.} Hoeffding as in Theorem~\ref{thm:aud-B} at level $\delta/3$
controls $|\widehat\gamma_{\mathrm{cal}}-\gamma(P_{\mathrm{cal}})|\le t(n_{\mathrm{lab}},\delta/3)$.
(iv)~\emph{Union.} On the intersection of the three events,
$|\gamma(P_T)|\le|\widehat\gamma_{\mathrm{cal}}|+t+\Delta_{\mathcal F}(\widehat\mu_{\mathrm{cal}},\widehat\mu_T)
+2\mathfrak R_n+2\mathfrak R_m+$concentration, which is $\widehat\beta_{\mathcal F}$.
Specialization~(i) follows because $W_1(\theta\#P,\theta\#Q)=\sup\{|Pf-Qf|:f(x)=h(\theta^\top\psi(x)),\,\mathrm{Lip}(h)\le1\}$
and the halfspace thresholds that realize the CDF form of $1$-d $W_1$ have VC dimension
$\le d+1$, recovering Theorem~\ref{thm:aud-F}'s $\varepsilon_{\mathrm{VC}}$ scale.
Specialization~(iii) is the classical empirical-$W_1$ rate on bounded domains.
\end{proof}

\begin{theorem}[Aud-G: domain-level verifiable alignment with an exact floor]\label{thm:aud-G}
Assume \textup{(G1)} the $K$ calibration domains and the deployment domain are exchangeable
tuples $(P_k,Z_k,\gamma_k)$; \textup{(G2)} each calibration domain carries a labeled sample of
\emph{common} size $n_{\mathrm{lab}}$, giving $\widehat\gamma_k$ with common Hoeffding radius
$e$ at level $\delta$; \textup{(G3)} the label-free drift predictor
$\gamma_{\mathrm{pred}}(\cdot)$ is fixed and independent of all $K{+}1$ domains. Let
$s_k=|\widehat\gamma_k-\gamma_{\mathrm{pred}}(Z_k)|$ and
$j=\lceil(1-\alpha+\delta)(K+1)\rceil$, feasible iff $\alpha\ge\tfrac1{K+1}+\delta$. Then
\[
\Pr\Bigl(\,|\gamma_{K+1}|\le|\gamma_{\mathrm{pred}}(Z_{K+1})|+s_{(j)}+e\,\Bigr)\ \ge\ 1-\alpha ,
\]
with every right-hand quantity computable without deployment labels.
\end{theorem}
\begin{proof}
The computable scores $s_1,\dots,s_{K+1}$ are exchangeable (common $n_{\mathrm{lab}}$ and
estimator under (G1)--(G3); if per-domain labeled sizes differ, exchangeability of the scores
fails and validity is lost). Split conformal at level $\alpha'=\alpha-\delta$ gives
$\Pr(s_{K+1}\le s_{(j)})\ge1-\alpha'$. One deployment-side Hoeffding event bounds the ideal
score $S_{K+1}=|\gamma_{K+1}-\gamma_{\mathrm{pred}}(Z_{K+1})|\le s_{K+1}+e$ (the counterfactual
size-$n_{\mathrm{lab}}$ sample interpretation of $e$); the deterministic triangle
$|\gamma_{K+1}|\le|\gamma_{\mathrm{pred}}(Z_{K+1})|+S_{K+1}$ and a union bound finish.
\emph{Floor:} the conformal index requires $j\le K$, i.e.\ $\alpha\ge\tfrac1{K+1}+\delta$
exactly; with $K=5$ domains nothing below $\alpha\approx0.17{+}\delta$ is certifiable, with
$K=8$ nothing below $\approx0.11{+}\delta$. Per-deployment verification remains impossible
(Theorem~\ref{thm:aud-A}); verifiability is \emph{priced in exchangeable domains}. (G3) is
essential: a predictor fit on the calibration domains breaks exchangeability.
\end{proof}

\begin{conjecture}[Maximal auditable class]\label{conj:aud-maximal}
Characterize the largest rotation-closed class $\mathcal U$
(Definition~\ref{def:rotation-closed}) that admits a \emph{useful}
(Definition~\ref{def:useful-audit}) label-free-at-deployment audit at rate
$O(\sqrt{d/n})$.
\paragraph{Proved bracket.} Useful audits at that rate exist whenever the dual witness class
has VC/Rademacher complexity $O(\sqrt{d/n})$ (Theorem~\ref{thm:aud-H}), including the
rotation-closed unknown-direction / bounded-$k$-index ridge classes of
Theorem~\ref{thm:aud-F}. Full Lipschitz / unrestricted $W_1$ witness classes are excluded from
useful $O(\sqrt{d/n})$ audits by the $n^{-1/d}$ empirical-$W_1$ rates on bounded
domains~\cite{fournier2015rate,weed2019sharp}.
\paragraph{Why estimation lower bounds do not finish the question.}
One-sided upper certificates (audits) and two-sided estimation are not equivalent: for
Wasserstein geometry on the cube, identity \emph{testing} can require roughly the square root of
the samples needed for estimation~\cite{deng2017wasserstein}. Thus $W_1$ estimation lower
bounds alone cannot prove that no intermediate rotation-closed class admits a useful root-$n$
audit.
\paragraph{Open.} Whether there exists a rotation-closed class strictly larger than
finite-index / adaptive MSW$_1$-ridge (e.g.\ slowly growing $k(n)$, constrained low-rank
potentials) that still admits a useful root-$n$ audit; and whether one-sided audits can beat
the spiked-transport estimation boundary~\cite{niles2019estimation}. A resolution plausibly
requires a new lower-bound construction over direction mixtures that targets useful UCBs
rather than two-sided estimators.
\end{conjecture}

\begin{conjecture}[Certified supremum, polynomial time]\label{conj:aud-computational}
Is there a $\mathrm{poly}(d,n)$-time algorithm that, from unlabeled samples, returns a number
$U$ satisfying both
\textup{(i)}~\emph{soundness:}
$\sup_{\theta}W_1(\theta\#\widehat\mu_{\mathrm{cal}},\theta\#\widehat\mu_T)\le U$ with
probability $\ge2/3$, and
\textup{(ii)}~\emph{null-completeness:} $U=O(\sqrt{d/n})$ with probability $\ge2/3$ whenever
$\mu_{\mathrm{cal}}=\mu_T$ (bounded support)?
\paragraph{Status.} Open for \emph{linear} max-sliced $W_1$. The strongest nearby evidence is
the statistical-query gap for spiked transport~\cite{niles2019estimation}; exact NP-hardness is
known for a \emph{kernel} max-sliced $W_2$ variant~\cite{wang2024kernelmsw} and does
\emph{not} transfer to the linear MSW$_1$ certificate required by Aud-F. Stationarity
guarantees for 1-d max-slicing exist~\cite{nietert2022sliced} but are not global certificates.
\paragraph{Operational rule.} Until settled, gradient-ascent surrogates for
$\mathrm{MS}_{\mathrm{net}}$ are heuristics; only the net value (or an equally certified
relaxation) may enter $\widehat\beta_F$ in Theorem~\ref{thm:aud-F}.
\end{conjecture}
